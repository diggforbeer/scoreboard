34def resolve_timezone(name: str, fallback: ZoneInfo | None = None) -> ZoneInfo: 

35 """``ZoneInfo`` for ``name``, degrading instead of raising on a bad value. 

36 

37 A typo in ``scoreboard.toml``'s ``timezone`` (e.g. "America/Chicagoo") 

38 must not crash the board -- see CLAUDE.md's config-conventions section 

39 and #80. On an unknown zone, this logs a warning and returns 

40 ``fallback`` if one was given (the currently running zone, so a bad 

41 live-reload keeps the board on whatever already worked) or else 

42 ``DEFAULT_TIMEZONE`` (boot and ``--dump`` have no previous value to 

43 keep). 

44 """ 

45 try: 

46 return ZoneInfo(name) 

47 except (ZoneInfoNotFoundError, ValueError): 

48 result = fallback if fallback is not None else ZoneInfo(DEFAULT_TIMEZONE) 

49 log.warning("Unknown timezone %r; using %s", name, result.key) 

50 return result

358@dataclass(slots=True) 

359class RotationEntry: 

360 """One slot in the configurable idle rotation (#150), parsed from a ``[[rotation]]`` table. 

361 

362 Only ever constructed by ``_parse_rotation`` with an already-validated 

363 ``screen``/``seconds`` pair -- invalid entries (unknown screen, 

364 non-positive seconds) are dropped there and never reach this type. 

365 """ 

366 

367 screen: str 

368 seconds: float 

369 

370 

371def _parse_rotation( 

372 raw: list[Any], 

373 valid: tuple[str, ...] | None = None, 

374 label: str = "[[rotation]]", 

375) -> list[RotationEntry]: 

376 """Parse ``[[rotation]]`` (an array of tables), dropping invalid entries. 

377 

378 ``valid`` and ``label`` let a caller use the same rules with its own screen names. 

379 

380 Per #150's decision 3, an invalid entry is never displayed on the panel 

381 but is not fatal either -- logged as a warning and skipped, same 

382 typo-tolerant convention as every other config value in this module. 

383 """ 

384 valid = VALID_ROTATION_SCREENS if valid is None else valid 

385 entries: list[RotationEntry] = [] 

386 for item in raw: 

387 if isinstance(item, RotationEntry): 

388 entries.append(item) 

389 continue 

390 if not isinstance(item, dict): 

391 log.warning("Ignoring %s entry that isn't a table: %r", label, item) 

392 continue 

393 screen = str(item.get("screen", "")).strip().lower() 

394 screen = ROTATION_SCREEN_ALIASES.get(screen, screen) 

395 if screen not in valid: 

396 log.warning( 

397 "Ignoring %s entry with unknown screen %r (must be one of %s)", 

398 label, 

399 item.get("screen"), 

400 ", ".join(valid), 

401 ) 

402 continue 

403 seconds = item.get("seconds") 

404 try: 

405 seconds_value = float(seconds) 

406 except (TypeError, ValueError): 

407 log.warning("Ignoring %s entry %r with non-numeric seconds %r", label, screen, seconds) 

408 continue 

409 if seconds_value <= 0: 

410 log.warning( 

411 "Ignoring %s entry %r with non-positive seconds %r", label, screen, seconds_value 

412 ) 

413 continue 

414 entries.append(RotationEntry(screen=screen, seconds=seconds_value)) 

415 return entries

795 return team in TEAM_COLORS if sport == "hockey" else football_team_exists(team)

798def _parse_teams(raw: Any) -> list[TeamEntry]: 

799 """The ranked list from ``[[teams]]``: validated per sport, in file order (rank 1 first). 

800 

801 A bad entry (not a table, an unknown sport, a team the sport does not have) is dropped with a 

802 warning rather than refusing to start; a repeat of a (sport, team) keeps its first, highest, 

803 place; anything past ``TEAMS_MAX`` is dropped. 

804 """ 

805 if raw in (None, []): 

806 return [] 

807 if not isinstance(raw, list): 

808 log.warning("teams must be a list of [[teams]] tables, got %r; ignoring it", raw) 

809 return [] 

810 teams: list[TeamEntry] = [] 

811 for index, row in enumerate(raw, start=1): 

812 if not isinstance(row, dict): 

813 log.warning("Ignoring teams entry %d: not a table", index) 

814 continue 

815 sport = str(row.get("sport", "")).strip().lower() 

816 team = str(row.get("team", "")).strip().upper() 

817 if sport not in TEAM_SPORTS: 

818 log.warning( 

819 "Ignoring teams entry %d: sport %r is not one of %s", index, sport, TEAM_SPORTS 

820 ) 

821 continue 

822 if not team_exists(sport, team): 

823 log.warning("Ignoring teams entry %d: %r is not a %s team", index, team, sport) 

824 continue 

825 entry = TeamEntry(sport, team) 

826 if entry in teams: 

827 log.warning( 

828 "Ignoring teams entry %d: %s %s is already ranked higher", index, sport, team 

829 ) 

830 continue 

831 if len(teams) >= TEAMS_MAX: 

832 log.warning( 

833 "Ignoring teams entry %d: only the first %d teams are kept", index, TEAMS_MAX 

834 ) 

835 continue 

836 teams.append(entry) 

837 return teams

916 def save(self, updates: Mapping[str, Any]) -> None: 

917 """Write ``updates`` into the source file, in place, keeping everything else. 

918 

919 ``updates`` is normally ``{section: {key: value}}``, e.g. 

920 ``{"scoreboard": {"rotate_seconds": 10}, "panel": {"brightness": 80}}`` 

921 -- only those keys are touched, parsed and re-emitted with 

922 ``tomlkit`` rather than ``tomllib`` + a plain writer, so every 

923 comment in the heavily-annotated boot-partition template survives 

924 untouched (#51). 

925 

926 The one exception is the ``"rotation"`` key (#150): ``[[rotation]]`` 

927 is an array of tables, not a flat section of scalars, so its value 

928 is the *whole new list* of ``{"screen": ..., "seconds": ...}`` 

929 dicts to write, not a ``{key: value}`` patch -- there is no 

930 per-field update for an ordered, variable-length list, only 

931 "replace it". Built with ``tomlkit.aot()`` rather than assigning a 

932 plain list, which would round-trip as an inline array rather than 

933 a sequence of ``[[rotation]]`` tables. 

934 

935 Updates this object's in-memory settings from the same file afterwards, 

936 via the normal load path, so the caller sees the merged result without 

937 a separate reload. 

938 """ 

939 if self.source_path is None: 

940 raise ConfigWriteError("no source file loaded; nothing to save to") 

941 try: 

942 doc = tomlkit.parse(self.source_path.read_text()) 

943 except OSError as exc: 

944 raise ConfigWriteError(f"could not read {self.source_path}: {exc}") from exc 

945 

946 for section, values in updates.items(): 

947 if section == "teams": 

948 # ``[[teams]]`` is the whole new ranked list, replaced like ``[[rotation]]``; an 

949 # empty list removes the key so no empty array is left behind. 

950 if not values: 

951 if "teams" in doc: 

952 del doc["teams"] 

953 continue 

954 aot = tomlkit.aot() 

955 for entry in values: 

956 row = tomlkit.table() 

957 row["sport"] = entry["sport"] 

958 row["team"] = entry["team"] 

959 aot.append(row) 

960 doc["teams"] = aot 

961 continue 

962 if section == "rotation": 

963 aot = tomlkit.aot() 

964 for entry in values: 

965 table = tomlkit.table() 

966 table["screen"] = entry["screen"] 

967 table["seconds"] = entry["seconds"] 

968 aot.append(table) 

969 doc["rotation"] = aot 

970 continue 

971 table = doc.get(section) 

972 if table is None: 

973 table = tomlkit.table() 

974 doc[section] = table 

975 for key, value in values.items(): 

976 table[key] = value 

977 for key in RETIRED_KEYS.get(section, {}): 

978 if key in table: 

979 del table[key] 

980 

981 tmp_path = self.source_path.with_name(self.source_path.name + ".tmp") 

982 try: 

983 tmp_path.write_text(tomlkit.dumps(doc)) 

984 tmp_path.replace(self.source_path) 

985 except OSError as exc: 

986 raise ConfigWriteError(f"could not write {self.source_path}: {exc}") from exc 

987 

988 reloaded = Settings.from_toml(self.source_path) 

989 self.panel = reloaded.panel 

990 self.scoreboard = reloaded.scoreboard 

991 self.football = reloaded.football 

992 self.teams = reloaded.teams 

993 self.rotation = reloaded.rotation 

994 self.audio = reloaded.audio 

995 self.status = reloaded.status 

996 self.wifi_setup = reloaded.wifi_setup 

997 self.wifi = reloaded.wifi 

998 self.night_mode = reloaded.night_mode 

999 self.update = reloaded.update 

1000 self.button = reloaded.button 

1001 self.holiday = reloaded.holiday

1004#: Settings a release has retired, by TOML section, with what replaced them 

1005#: (#277). Unlike a typo they're expected on an upgraded board, so they're

1007#: ``Settings.save`` drops them the next time their section is saved. 

1008RETIRED_KEYS: dict[str, dict[str, str]] = { 

1009 "scoreboard": { 

1010 "sport": "replaced by the ranked [[teams]] list (the sport follows your top team)", 

1011 "favourite_team": "replaced by the ranked [[teams]] list (add your team on the admin page)", 

1012 "also_follow": "replaced by the ranked [[teams]] list (add your teams on the admin page)",

1016 "favourite_team": "replaced by the ranked [[teams]] list (add your team on the admin page)", 

1017 "also_follow": "replaced by the ranked [[teams]] list (add your teams on the admin page)",

1019 "holiday": { 

1020 "flyby_min_minutes": "replaced by flyby_frequency", 

1021 "flyby_max_minutes": "replaced by flyby_frequency", 

1022 }, 

1023} 

1024 

1025#: (section, key) pairs already logged by _build this process. 

1026_retired_logged: set[tuple[str, str]] = set()

1029def _build(cls: type, raw: dict[str, Any], section: str = "") -> Any: 

1030 """Construct a dataclass from ``raw``, ignoring unknown keys. 

1031 

1032 Unknown keys are warned about rather than fatal: a user hand-editing the 

1033 file on the boot partition should never end up with a board that refuses 

1034 to start because of one typo. Retired keys (``RETIRED_KEYS``) are 

1035 ignored too, but logged only once. 

1036 """ 

1037 # init=False fields are derived (e.g. NightModeConfig.start), not settable. 

1038 known = {f.name for f in fields(cls) if f.init} 

1039 retired = RETIRED_KEYS.get(section, {}) 

1040 kwargs = {} 

1041 for key, value in raw.items(): 

1042 if key in known: 

1043 kwargs[key] = value 

1044 elif key in retired: 

1045 if (section, key) not in _retired_logged: 

1046 _retired_logged.add((section, key)) 

1047 # The team settings are the ones an owner will miss, so those say so loudly. 

1048 loud = "[[teams]]" in retired[key] or "[[rotation]]" in retired[key] 

1049 level = log.warning if loud else log.info 

1050 level( 

1051 "Ignoring retired config key %r in [%s] (%s); safe to delete, " 

1052 "and removed the next time [%s] is saved", 

1053 key, 

1054 section, 

1055 retired[key], 

1056 section, 

1057 ) 

1058 else: 

1059 log.warning("Ignoring unknown config key %r in [%s]", key, cls.__name__) 

1060 return cls(**kwargs)
