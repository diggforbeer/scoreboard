29def resolve_timezone(name: str, fallback: ZoneInfo | None = None) -> ZoneInfo: 

30 """``ZoneInfo`` for ``name``, degrading instead of raising on a bad value. 

31 

32 A typo in ``scoreboard.toml``'s ``timezone`` (e.g. "America/Chicagoo") 

33 must not crash the board -- see CLAUDE.md's config-conventions section 

34 and #80. On an unknown zone, this logs a warning and returns 

35 ``fallback`` if one was given (the currently running zone, so a bad 

36 live-reload keeps the board on whatever already worked) or else 

37 ``DEFAULT_TIMEZONE`` (boot and ``--dump`` have no previous value to 

38 keep). 

39 """ 

40 try: 

41 return ZoneInfo(name) 

42 except (ZoneInfoNotFoundError, ValueError): 

43 result = fallback if fallback is not None else ZoneInfo(DEFAULT_TIMEZONE) 

44 log.warning("Unknown timezone %r; using %s", name, result.key) 

45 return result

239def _parse_rotation(raw: list[Any]) -> list[RotationEntry]: 

240 """Parse ``[[rotation]]`` (an array of tables), dropping invalid entries. 

241 

242 Per #150's decision 3, an invalid entry is never displayed on the panel 

243 but is not fatal either -- logged as a warning and skipped, same 

244 typo-tolerant convention as every other config value in this module. 

245 """ 

246 entries: list[RotationEntry] = [] 

247 for item in raw: 

248 if not isinstance(item, dict): 248 ↛ 249line 248 didn't jump to line 249 because the condition on line 248 was never true

249 log.warning("Ignoring [[rotation]] entry that isn't a table: %r", item) 

250 continue 

251 screen = str(item.get("screen", "")).strip().lower() 

252 if screen not in VALID_ROTATION_SCREENS: 

253 log.warning( 

254 "Ignoring [[rotation]] entry with unknown screen %r (must be one of %s)", 

255 item.get("screen"), 

256 ", ".join(VALID_ROTATION_SCREENS), 

257 ) 

258 continue 

259 seconds = item.get("seconds") 

260 try: 

261 seconds_value = float(seconds) 

262 except (TypeError, ValueError): 

263 log.warning( 

264 "Ignoring [[rotation]] entry %r with non-numeric seconds %r", screen, seconds 

265 ) 

266 continue 

267 if seconds_value <= 0: 

268 log.warning( 

269 "Ignoring [[rotation]] entry %r with non-positive seconds %r", screen, seconds_value 

270 ) 

271 continue 

272 entries.append(RotationEntry(screen=screen, seconds=seconds_value)) 

273 return entries

499 def save(self, updates: Mapping[str, Any]) -> None: 

500 """Write ``updates`` into the source file, in place, keeping everything else. 

501 

502 ``updates`` is normally ``{section: {key: value}}``, e.g. 

503 ``{"scoreboard": {"favourite_team": "TOR"}, "panel": {"brightness": 80}}`` 

504 -- only those keys are touched, parsed and re-emitted with 

505 ``tomlkit`` rather than ``tomllib`` + a plain writer, so every 

506 comment in the heavily-annotated boot-partition template survives 

507 untouched (#51). 

508 

509 The one exception is the ``"rotation"`` key (#150): ``[[rotation]]`` 

510 is an array of tables, not a flat section of scalars, so its value 

511 is the *whole new list* of ``{"screen": ..., "seconds": ...}`` 

512 dicts to write, not a ``{key: value}`` patch -- there is no 

513 per-field update for an ordered, variable-length list, only 

514 "replace it". Built with ``tomlkit.aot()`` rather than assigning a 

515 plain list, which would round-trip as an inline array rather than 

516 a sequence of ``[[rotation]]`` tables. 

517 

518 Updates this object's in-memory settings from the same file afterwards, 

519 via the normal load path, so the caller sees the merged result without 

520 a separate reload. 

521 """ 

522 if self.source_path is None: 

523 raise ConfigWriteError("no source file loaded; nothing to save to") 

524 try: 

525 doc = tomlkit.parse(self.source_path.read_text()) 

526 except OSError as exc: 

527 raise ConfigWriteError(f"could not read {self.source_path}: {exc}") from exc 

528 

529 for section, values in updates.items(): 

530 if section == "rotation": 

531 aot = tomlkit.aot() 

532 for entry in values: 

533 table = tomlkit.table() 

534 table["screen"] = entry["screen"] 

535 table["seconds"] = entry["seconds"] 

536 aot.append(table) 

537 doc["rotation"] = aot 

538 continue 

539 table = doc.get(section) 

540 if table is None: 

541 table = tomlkit.table() 

542 doc[section] = table 

543 for key, value in values.items(): 

544 table[key] = value 

545 

546 tmp_path = self.source_path.with_name(self.source_path.name + ".tmp") 

547 try: 

548 tmp_path.write_text(tomlkit.dumps(doc)) 

549 tmp_path.replace(self.source_path) 

550 except OSError as exc: 

551 raise ConfigWriteError(f"could not write {self.source_path}: {exc}") from exc

566def _build(cls: type, raw: dict[str, Any]) -> Any: 

567 """Construct a dataclass from ``raw``, ignoring unknown keys. 

568 

569 Unknown keys are warned about rather than fatal: a user hand-editing the 

570 file on the boot partition should never end up with a board that refuses 

571 to start because of one typo. 

572 """ 

573 # init=False fields are derived (e.g. NightModeConfig.start), not settable. 

574 known = {f.name for f in fields(cls) if f.init} 

575 kwargs = {} 

576 for key, value in raw.items(): 

577 if key in known: 

578 kwargs[key] = value 

579 else: 

580 log.warning("Ignoring unknown config key %r in [%s]", key, cls.__name__) 

581 return cls(**kwargs)

584def _wifi_config(raw: dict[str, Any]) -> WifiConfig: 

585 """WifiConfig only models ``connect_timeout_seconds``. 

586 

587 ``ssid``/``password``/``country`` live in the same ``[wifi]`` TOML 

588 section but are read directly out of the raw dict by 

589 scoreboard-provision (predates this dataclass), not through here. 

590 Filtered before ``_build()`` sees them so those three don't trip its 

591 "unknown key" warning on every single load. 

592 """ 

593 filtered = {k: v for k, v in raw.items() if k == "connect_timeout_seconds"} 

594 return _build(WifiConfig, filtered)
