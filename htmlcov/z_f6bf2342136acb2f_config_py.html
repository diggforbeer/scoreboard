39@dataclass(slots=True) 

40class PanelConfig: 

41 """Geometry and electrical settings for the HUB75 chain. 

42 

43 Defaults describe two 64x32 panels daisy-chained into one 128x32 

44 display through a HUB75 adapter wired to the driver's "regular" pinout. 

45 Pixel pitch does not affect the driver; it is recorded so physical 

46 dimensions can be derived rather than remembered. 

47 """ 

48 

49 rows: int = 32 

50 cols: int = 64 

51 chain_length: int = 2 

52 parallel: int = 1 

53 pitch_mm: float = 2.5 

54 hardware_mapping: str = "regular" 

55 gpio_slowdown: int = 4 

56 pwm_bits: int = 11 

57 pwm_lsb_nanoseconds: int = 130 

58 #: Static brightness, and the fallback used whenever auto_brightness is 

59 #: off or the ambient sensor (#44) isn't answering. 

60 brightness: int = 60 

61 limit_refresh_rate_hz: int = 0 

62 disable_hardware_pulsing: bool = False 

63 #: rpi-rgb-led-matrix's --led-pixel-mapper equivalent, e.g. "U-mapper" 

64 #: to fold two panels chained end-to-end into a stacked 64x64 display, 

65 #: or "Rotate:180" / "Mirror:H" for a panel mounted flipped. Chain 

66 #: several with ";", e.g. "U-mapper;Rotate:90". Empty means none. 

67 pixel_mapper: str = "" 

68 

69 #: Dim/brighten the panel from a BH1750 ambient light sensor on the I2C 

70 #: bus instead of a fixed `brightness`. Off by default: not every board 

71 #: has the sensor wired up, and `LightSensor` degrades to `brightness` 

72 #: on its own if this is on but nothing answers, so there's no harm in 

73 #: leaving it on for a board without the sensor -- it's just extra I2C 

74 #: probing for nothing. 

75 auto_brightness: bool = False 

76 min_brightness: int = 10 

77 max_brightness: int = 100 

78 #: How often the sensor is sampled and brightness re-applied. Smoothed 

79 #: on top of this (see ScoreboardApp.refresh_brightness) so this can be 

80 #: fairly frequent without the panel visibly flickering. 

81 brightness_poll_seconds: float = 5.0

174@dataclass(slots=True) 

175class Settings: 

176 panel: PanelConfig = field(default_factory=PanelConfig) 

177 scoreboard: ScoreboardConfig = field(default_factory=ScoreboardConfig) 

178 audio: AudioConfig = field(default_factory=AudioConfig) 

179 status: StatusServerConfig = field(default_factory=StatusServerConfig) 

180 source_path: Path | None = None 

181 

182 @classmethod 

183 def load(cls, path: str | os.PathLike[str] | None = None) -> Settings: 

184 """Load settings from ``path``, the first default location, or defaults.""" 

185 candidates = [Path(path)] if path else list(DEFAULT_CONFIG_PATHS) 

186 for candidate in candidates: 

187 if candidate.is_file(): 

188 return cls.from_toml(candidate) 

189 log.info("No config file found in %s; using built-in defaults", candidates) 

190 return cls() 

191 

192 @classmethod 

193 def from_toml(cls, path: Path) -> Settings: 

194 with path.open("rb") as fh: 

195 raw = tomllib.load(fh) 

196 settings = cls.from_dict(raw) 

197 settings.source_path = path 

198 log.info("Loaded configuration from %s", path) 

199 return settings 

200 

201 @classmethod 

202 def from_dict(cls, raw: dict[str, Any]) -> Settings: 

203 return cls( 

204 panel=_build(PanelConfig, raw.get("panel", {})), 

205 scoreboard=_build(ScoreboardConfig, raw.get("scoreboard", {})), 

206 audio=_build(AudioConfig, raw.get("audio", {})), 

207 status=_build(StatusServerConfig, raw.get("status", {})), 

208 )

210 def save(self, updates: Mapping[str, Mapping[str, Any]]) -> None: 

211 """Write ``updates`` into the source file, in place, keeping everything else. 

212 

213 ``updates`` is ``{section: {key: value}}``, e.g. 

214 ``{"scoreboard": {"favourite_team": "TOR"}, "panel": {"brightness": 80}}``. 

215 Only those keys are touched -- parsed and re-emitted with ``tomlkit`` 

216 rather than ``tomllib`` + a plain writer, so every comment in the 

217 heavily-annotated boot-partition template survives untouched (#51). 

218 

219 Updates this object's in-memory settings from the same file afterwards, 

220 via the normal load path, so the caller sees the merged result without 

221 a separate reload. 

222 """ 

223 if self.source_path is None: 

224 raise ConfigWriteError("no source file loaded; nothing to save to") 

225 try: 

226 doc = tomlkit.parse(self.source_path.read_text()) 

227 except OSError as exc: 

228 raise ConfigWriteError(f"could not read {self.source_path}: {exc}") from exc 

229 

230 for section, values in updates.items(): 

231 table = doc.get(section) 

232 if table is None: 

233 table = tomlkit.table() 

234 doc[section] = table 

235 for key, value in values.items(): 

236 table[key] = value 

237 

238 tmp_path = self.source_path.with_name(self.source_path.name + ".tmp") 

239 try: 

240 tmp_path.write_text(tomlkit.dumps(doc)) 

241 tmp_path.replace(self.source_path) 

242 except OSError as exc: 

243 raise ConfigWriteError(f"could not write {self.source_path}: {exc}") from exc 

244 

245 reloaded = Settings.from_toml(self.source_path) 

246 self.panel = reloaded.panel 

247 self.scoreboard = reloaded.scoreboard 

248 self.audio = reloaded.audio 

249 self.status = reloaded.status

252def _build(cls: type, raw: dict[str, Any]) -> Any: 

253 """Construct a dataclass from ``raw``, ignoring unknown keys. 

254 

255 Unknown keys are warned about rather than fatal: a user hand-editing the 

256 file on the boot partition should never end up with a board that refuses 

257 to start because of one typo. 

258 """ 

259 known = {f.name for f in fields(cls)} 

260 kwargs = {} 

261 for key, value in raw.items(): 

262 if key in known: 

263 kwargs[key] = value 

264 else: 

265 log.warning("Ignoring unknown config key %r in [%s]", key, cls.__name__) 

266 return cls(**kwargs)
