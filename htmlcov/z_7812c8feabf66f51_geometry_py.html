1"""The maths every field screen shares (#358): where a yard is, which way a team drives, 

2where the first-down line is, which lane a run or pass takes, and the arc of a kick. 

3 

4Pure functions and one small layout class, no drawing and no Pillow. The screens (the field, 

5the run, the pass, the kicks) all use these, so they cannot disagree about where the ball is. 

6 

7**The one thing to know about ESPN's ``yardLine``** (``football_api.md``, "Field coordinates"): 

8it runs 0-100 **from the home team's goal line**, not from the offence's. So the home end zone is 

9always on the left and the away end zone on the right, the home team drives **toward** 100 (to 

10the right) and the away team toward 0 (to the left). "NYG 28" is 28 whether NYG has the ball or 

11not; "WSH 20" is 80. Which way the first-down line lies, and any left/right flip, follow from 

12who has the ball, never from the yard line itself. 

13 

14**100 yards are 100 pixels** on a full 128px panel: with 14px end zones the playing field is 

15exactly 100px wide, one pixel a yard. On a 64px panel (or a 64px region of a wider one) the end 

16zones are 7px and the field is about 50px, half a pixel a yard. 

17"""

27# -- layout constants. Mirrored in tests/test_football_field.py: change both together. --------- 

28#: Rows (inclusive) the field occupies. Rows above hold the one-line band of text. 

29FIELD_TOP = 8 

30FIELD_BOTTOM = 30 

31#: End-zone depth in pixels: 14 gives a 100px field on a 128px panel, 7 on anything narrower. 

32EZ_WIDE = 14 

33EZ_NARROW = 7 

34#: A field region this wide or wider gets the wide end zones. 

35WIDE_MIN_WIDTH = 112 

36#: Rows kept clear at the top and bottom edge of the field for the lanes. 

37LANE_MARGIN = 2 

38 

39HOME = "home" 

40AWAY = "away" 

41 

42#: How far across the field (0 = the offence's left, 1 = its right) each run gap and pass 

43#: direction lies. The offence faces the goal it is attacking, so its left is up the panel when 

44#: it drives right and down the panel when it drives left (see ``lane_y``). 

45LANES: dict[str, float] = { 

46 "left end": 0.10, 

47 "left tackle": 0.22, 

48 "left guard": 0.36, 

49 "up the middle": 0.50, 

50 "right guard": 0.64, 

51 "right tackle": 0.78, 

52 "right end": 0.90, 

53 # A pass is thrown to a side, not a gap. 

54 "left": 0.20, 

55 "middle": 0.50, 

56 "right": 0.80, 

57 # The empty string (the text named no direction) goes up the middle. 

58 "": 0.50, 

59}

62@dataclass(frozen=True, slots=True) 

63class FieldLayout: 

64 """Where the field sits on the panel (or in a region of it).""" 

65 

66 left: int #: first column of the region 

67 right: int #: one past the last column 

68 top: int = FIELD_TOP 

69 bottom: int = FIELD_BOTTOM #: the last row, inclusive 

70 

71 @classmethod 

72 def for_panel(cls, width: int, left: int = 0, right: int | None = None) -> FieldLayout: 

73 """The whole panel, or the region ``left``..``right`` of it (the play-text layout puts 

74 the field in the left half).""" 

75 return cls(left=left, right=width if right is None else right) 

76 

77 # -- the region ----------------------------------------------------- 

78 

79 @property 

80 def width(self) -> int: 

81 return self.right - self.left 

82 

83 @property 

84 def end_zone(self) -> int: 

85 """End-zone depth in pixels: 14 on a wide field (100px for 100 yards), else 7.""" 

86 return EZ_WIDE if self.width >= WIDE_MIN_WIDTH else EZ_NARROW 

87 

88 @property 

89 def x0(self) -> int: 

90 """The column of yard line 0, the home goal line.""" 

91 return self.left + self.end_zone 

92 

93 @property 

94 def span(self) -> int: 

95 """Pixels from goal line to goal line.""" 

96 return self.width - 2 * self.end_zone 

97 

98 @property 

99 def height(self) -> int: 

100 return self.bottom - self.top + 1 

101 

102 @property 

103 def mid_y(self) -> int: 

104 return (self.top + self.bottom) // 2 

105 

106 @property 

107 def pixels_per_yard(self) -> float: 

108 return self.span / 100 

109 

110 # -- yards and pixels ---------------------------------------------------- 

111 

112 def x(self, yard_line: float) -> float: 

113 """The (fractional) column of ``yard_line``, 0-100 from the home goal line.""" 

114 return self.x0 + self.span * yard_line / 100 

115 

116 def px(self, yard_line: float) -> int: 

117 return round(self.x(yard_line)) 

118 

119 def yard_at(self, x: float) -> float: 

120 """The yard line at column ``x``: the inverse of :meth:`x`.""" 

121 return (x - self.x0) * 100 / self.span 

122 

123 def end_zone_columns(self, side: str) -> tuple[int, int]: 

124 """First and last column (inclusive) of ``HOME``'s or ``AWAY``'s end zone.""" 

125 if side == HOME: 

126 return self.left, self.x0 - 1 

127 if side == AWAY: 

128 return self.px(100), self.right - 1 

129 raise ValueError(f"side must be {HOME!r} or {AWAY!r}, not {side!r}")

135def drive_direction(possession: str) -> int: 

136 """``+1`` if the team with the ball drives toward yard line 100 (the home team does, to the 

137 right), ``-1`` toward 0 (the away team, to the left).""" 

138 if possession == HOME: 

139 return 1 

140 if possession == AWAY: 

141 return -1 

142 raise ValueError(f"possession must be {HOME!r} or {AWAY!r}, not {possession!r}") 

143 

144 

145def clamp_yard_line(yard_line: float) -> float: 

146 return max(0.0, min(100.0, yard_line)) 

147 

148 

149def first_down_line(yard_line: float, distance: float, direction: int) -> float: 

150 """Where the first-down marker is: ``distance`` yards ahead of the ball in the direction of 

151 play, stopping at the goal line (in goal-to-go the marker is the goal line itself).""" 

152 return clamp_yard_line(yard_line + direction * distance) 

153 

154 

155def goal_to_go(yard_line: float, distance: float, direction: int) -> bool: 

156 """True when the first-down line would be past the goal line.""" 

157 target = yard_line + direction * distance 

158 return target <= 0 or target >= 100 

159 

160 

161def possession_side(game: FbGame, team_id: int | None) -> str | None: 

162 """``HOME``, ``AWAY`` or ``None`` for the team with the ball, from its ESPN id.""" 

163 if team_id is None: 

164 return None 

165 if game.home.espn_id == team_id: 

166 return HOME 

167 if game.away.espn_id == team_id: 

168 return AWAY 

169 return None

175def lane_y(lane: str | float, direction: int, layout: FieldLayout) -> float: 

176 """The row of a run gap or pass direction (``"left tackle"``, ``"middle"``, or a fraction). 

177 

178 The offence faces the goal it attacks, so its *left* is the top of the panel when it drives 

179 right (``direction`` +1) and the bottom when it drives left. A name the table does not know 

180 goes up the middle rather than raising: the text is from an undocumented feed. 

181 """ 

182 fraction = lane if isinstance(lane, float | int) else LANES.get(lane, 0.5) 

183 fraction = max(0.0, min(1.0, float(fraction))) 

184 if direction < 0: 

185 fraction = 1.0 - fraction 

186 low = layout.top + LANE_MARGIN 

187 high = layout.bottom - LANE_MARGIN 

188 return low + fraction * (high - low) 

189 

190 

191def arc_height(yards: float, minimum: float = 3.0, maximum: float = 14.0) -> float: 

192 """How high a kick or pass of ``yards`` arcs, in pixels: longer goes higher, within limits.""" 

193 return max(minimum, min(maximum, abs(yards) * 0.22)) 

194 

195 

196def arc_point( 

197 start_x: float, end_x: float, base_y: float, height: float, t: float 

198) -> tuple[float, float]: 

199 """The ball at fraction ``t`` (0-1) of a parabolic arc from ``start_x`` to ``end_x``, rising 

200 ``height`` pixels above ``base_y`` at the middle and back to ``base_y`` at the ends.""" 

201 t = max(0.0, min(1.0, t)) 

202 return start_x + (end_x - start_x) * t, base_y - height * 4 * t * (1 - t) 

203 

204 

205def arc_points( 

206 start_x: float, end_x: float, base_y: float, height: float, steps: int | None = None 

207) -> list[tuple[float, float]]: 

208 """Points along the arc, about one per pixel of horizontal travel.""" 

209 n = steps if steps is not None else max(8, int(abs(end_x - start_x))) 

210 return [arc_point(start_x, end_x, base_y, height, i / n) for i in range(n + 1)]
