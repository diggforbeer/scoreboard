98 

99 ``skip_left``/``skip_right`` drop that many source columns from the 

100 logo's centre-facing edge, so it can stay flush against the panel's 

101 outer edge while cropped -- see ``_logo_span`` (#38).

104 for dx, dy, (r, g, b) in logo.pixels:

108 #: Minimum width reserved for the score/divider/status column between 

109 #: the two logos. A 128-wide panel (two chained 64x32 panels) has ample 

110 #: room and this never binds -- both logos draw in full, same as always. 

111 #: A single 64x32 panel doesn't: two full 32px logos would leave zero 

112 #: room for scores, so each logo crops its centre-facing edge instead of 

113 #: shrinking, down to half its width and no further (#38).

116 def _logo_span(self, away: Logo, home: Logo) -> tuple[int, int, int, int]: 

117 """How much of each logo to show, and the resulting middle column. 

118 

119 Returns ``(away_visible, home_visible, left, right)``. Each logo 

120 stays flush against its panel edge; only its centre-facing edge is 

121 cropped, so what's lost is the part nearest the score column, not 

122 the recognisable outer silhouette. 

123 """ 

124 per_side = (self.width - self._LOGO_MIN_MIDDLE) // 2 

125 away_visible = max(away.width // 2, min(away.width, per_side)) 

126 home_visible = max(home.width // 2, min(home.width, per_side)) 

127 return away_visible, home_visible, away_visible, self.width - home_visible

268 #: Shootout grid (#259): a row per team (away on top), a column per 

269 #: attempt. Cells are 9px discs/crosses on an 11px pitch. 

270 _SO_ROW_CENTRES = (8, 20) 

271 _SO_CELL_RADIUS = 4 

272 _SO_CELL_PITCH = 11 

273 _SO_CELLS_LEFT = 34 

274 _SO_MIN_CELLS = 3 

275 #: Newest-attempt animation, in seconds since the attempt first showed 

276 #: up. SHOOTOUT_ANIM_SECONDS is the whole window (the app runs its fast 

277 #: loop for exactly that long); a goal grows 0-0.9s then flashes a 

278 #: white ring until 1.3s, a miss draws its two strokes over 0-1.2s. 

279 SHOOTOUT_ANIM_SECONDS = 1.5 

280 _SO_GROW_SECONDS = 0.9 

281 _SO_FLASH_SECONDS = 1.3 

282 _SO_STROKE_SECONDS = 0.6 

283 #: How long, and how fast, a changed tally blinks. 

284 _SO_TALLY_FLASH_SECONDS = 1.0 

285 _SO_TALLY_BLINK_SECONDS = 0.25 

286 #: Period of the "must score" outline's slow breathe. 

287 _SO_PULSE_SECONDS = 3.0

289 def draw_shootout( 

290 self, 

291 canvas: Any, 

292 game: Game, 

293 now: float = 0.0, 

294 ages: Mapping[int, float] | None = None, 

295 ) -> None: 

296 """Away row over home row, one cell per attempt: green disc = goal, red X = miss. 

297 

298 Cells not yet taken are dim rings; the grid grows past three in 

299 sudden death and, once it outgrows the panel, shows the most recent 

300 attempts that fit. Rows are found by abbreviation (the first shooter 

301 varies), so a team with one more attempt than the other just has one 

302 more filled cell. No logos: the grid needs the width. 

303 

304 ``ages`` maps an attempt's ``sequence`` to seconds since it first 

305 appeared; attempts younger than ``SHOOTOUT_ANIM_SECONDS`` animate, 

306 everything else (and ``ages=None``) is the static look. ``now`` is a 

307 monotonic time that only drives the "must score" outline's breathe. 

308 The bottom label follows ``Shootout.state()``: ``MUST SCORE`` / 

309 ``SUDDEN DEATH``, else ``SHOOTOUT``, or ``FINAL/SO`` once over. 

310 """ 

311 canvas.Clear() 

312 shootout = game.shootout 

313 finished = game.is_final 

314 ages = ages or {} 

315 winner = None 

316 if finished and shootout is not None and shootout.away_goals != shootout.home_goals: 

317 winner = "away" if shootout.away_goals > shootout.home_goals else "home" 

318 state = ( 

319 shootout.state(game.away.abbrev, game.home.abbrev) 

320 if shootout is not None 

321 else ShootoutState("open") 

322 ) 

323 

324 sides = (("away", game.away), ("home", game.home)) 

325 results = [ 

326 shootout.results_for(side.abbrev) if shootout is not None else () for _, side in sides 

327 ] 

328 goals = (shootout.away_goals, shootout.home_goals) if shootout is not None else (0, 0) 

329 capacity = max(1, (self.width - self._SO_CELLS_LEFT - 2) // self._SO_CELL_PITCH) 

330 rounds = max(self._SO_MIN_CELLS, *(len(r) for r in results)) 

331 first = max(0, rounds - capacity) 

332 

333 for (key, side), row, total, centre_y in zip( 

334 sides, results, goals, self._SO_ROW_CENTRES, strict=True 

335 ): 

336 font = self.fonts.small 

337 baseline = centre_y + 3 

338 row_ages = self._attempt_ages(shootout, side.abbrev, ages) 

339 self.text(canvas, font, 2, baseline, team_color(side.abbrev), side.abbrev) 

340 tally_colour = ACCENT if winner == key else WHITE 

341 if self._tally_flashing(shootout, side.abbrev, ages): 

342 tally_colour = ACCENT if winner != key else WHITE 

343 self.text_right( 

344 canvas, font, self._SO_CELLS_LEFT - 5, baseline, tally_colour, str(total) 

345 ) 

346 for column, index in enumerate(range(first, min(rounds, first + capacity))): 

347 cx = self._SO_CELLS_LEFT + self._SO_CELL_RADIUS + column * self._SO_CELL_PITCH 

348 if index >= len(row): 

349 self._draw_cell_pending(canvas, cx, centre_y) 

350 if ( 

351 state.kind == "must_score" 

352 and state.team == side.abbrev 

353 and index == len(row) 

354 ): 

355 self._draw_cell_outline(canvas, cx, centre_y, now) 

356 continue 

357 age = row_ages[index] 

358 if row[index]: 

359 self._draw_cell_goal(canvas, cx, centre_y, age) 

360 else: 

361 self._draw_cell_miss(canvas, cx, centre_y, age) 

362 

363 if finished: 

364 label, colour = "FINAL/SO", FINAL 

365 elif state.kind == "must_score": 

366 label, colour = "MUST SCORE", INTERMISSION 

367 elif state.kind == "sudden_death": 

368 label, colour = "SUDDEN DEATH", INTERMISSION 

369 else: 

370 label, colour = "SHOOTOUT", LIVE 

371 self.text_center(canvas, self.fonts.tiny, self.width // 2, self.height - 1, colour, label)

374 def _attempt_ages( 

375 shootout: Shootout | None, abbrev: str, ages: Mapping[int, float] 

376 ) -> list[float | None]: 

377 """Age of each of ``abbrev``'s attempts in order, None once static.""" 

378 if shootout is None: 

379 return []

382 def _tally_flashing( 

383 self, shootout: Shootout | None, abbrev: str, ages: Mapping[int, float] 

384 ) -> bool: 

385 """Blink the tally while ``abbrev``'s newest goal is still fresh.""" 

386 if shootout is None: 

387 return False 

388 for a in shootout.attempts: 

389 age = ages.get(a.sequence) 

390 fresh = age is not None and 0 <= age < self._SO_TALLY_FLASH_SECONDS 

391 if a.team_abbrev == abbrev and a.scored and fresh: 

392 return int(age / self._SO_TALLY_BLINK_SECONDS) % 2 == 0 

393 return False

395 def _draw_cell_goal(self, canvas: Any, cx: int, cy: int, age: float | None = None) -> None: 

396 r = self._SO_CELL_RADIUS 

397 if age is None or age >= self.SHOOTOUT_ANIM_SECONDS: 

398 grow = 1.0 

399 else: 

400 grow = min(1.0, max(0.0, age) / self._SO_GROW_SECONDS) 

401 for dy in range(-r, r + 1): 

402 for dx in range(-r, r + 1): 

403 if dx * dx + dy * dy <= (r * grow) ** 2 + 2 * grow: 

404 canvas.SetPixel(cx + dx, cy + dy, *LIVE) 

405 if age is not None and self._SO_GROW_SECONDS <= age < self._SO_FLASH_SECONDS: 

406 self._draw_ring(canvas, cx, cy, r + 1, WHITE)

408 def _draw_ring(self, canvas: Any, cx: int, cy: int, radius: int, rgb) -> None: 

409 lo, hi = (radius - 0.5) ** 2, (radius + 0.5) ** 2 

410 for dy in range(-radius, radius + 1): 

411 for dx in range(-radius, radius + 1): 

412 if lo <= dx * dx + dy * dy <= hi: 

413 canvas.SetPixel(cx + dx, cy + dy, *rgb)

415 def _draw_cell_outline(self, canvas: Any, cx: int, cy: int, now: float) -> None: 

416 """The "must score" ring around a pending cell, breathing slowly.""" 

417 phase = (now % self._SO_PULSE_SECONDS) / self._SO_PULSE_SECONDS 

418 level = 0.55 + 0.45 * (0.5 + 0.5 * math.sin(2 * math.pi * phase)) 

419 rgb = tuple(round(c * level) for c in INTERMISSION) 

420 self._draw_ring(canvas, cx, cy, self._SO_CELL_RADIUS + 1, rgb)

422 def _draw_cell_pending(self, canvas: Any, cx: int, cy: int) -> None: 

423 r = self._SO_CELL_RADIUS 

424 for dy in range(-r, r + 1): 

425 for dx in range(-r, r + 1): 

426 if (r - 1) ** 2 <= dx * dx + dy * dy <= r * r + 2: 

427 canvas.SetPixel(cx + dx, cy + dy, *SUBDUED)

429 def _draw_cell_miss(self, canvas: Any, cx: int, cy: int, age: float | None = None) -> None: 

430 """The red X; while fresh its two strokes are drawn one after the other.""" 

431 reach = self._SO_CELL_RADIUS - 1 

432 span = 2 * reach + 1 

433 if age is None or age >= self.SHOOTOUT_ANIM_SECONDS: 

434 first = second = 1.0 

435 else: 

436 stroke = self._SO_STROKE_SECONDS 

437 first = min(1.0, max(0.0, age) / stroke) 

438 second = min(1.0, max(0.0, age - stroke) / stroke) 

439 for i in range(-reach, -reach + math.ceil(span * first)): 

440 canvas.SetPixel(cx + i, cy + i, *LOSS) 

441 for i in range(-reach, -reach + math.ceil(span * second)): 

442 canvas.SetPixel(cx + i, cy - i, *LOSS)
