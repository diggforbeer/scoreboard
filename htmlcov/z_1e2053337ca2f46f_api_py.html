1"""Client for the public NHL web API (api-web.nhle.com). 

2 

3This is the same undocumented-but-public API the NHL's own site uses. It is 

4unauthenticated and rate-limit friendly at the poll intervals we use, but it 

5is not contractual -- every call is wrapped so that a bad response degrades to 

6"no new data" rather than taking the board down. 

7""" 

8 

9from __future__ import annotations 

10 

11import logging 

12from collections.abc import Callable 

13from typing import Any 

14 

15import requests 

16from requests.adapters import HTTPAdapter 

17from urllib3.util.retry import Retry 

18 

19from .models import Game, Situation, StandingsRow 

20 

21log = logging.getLogger(__name__) 

22 

23BASE_URL = "https://api-web.nhle.com/v1" 

24USER_AGENT = "nhl-scoreboard/0.1 (+https://github.com/diggforbeer/scoreboard)" 

25

27class NHLApiError(RuntimeError): 

28 """Raised when the API could not be reached or returned unusable data.""" 

29 

30 

31class NHLClient: 

32 def __init__( 

33 self, 

34 base_url: str = BASE_URL, 

35 timeout: float = 10.0, 

36 session: requests.Session | None = None, 

37 ) -> None: 

38 self.base_url = base_url.rstrip("/") 

39 self.timeout = timeout 

40 self.session = session or _build_session() 

41 

42 def close(self) -> None: 

43 self.session.close() 

44 

45 def __enter__(self) -> NHLClient: 

46 return self

51 # -- endpoints ------------------------------------------------------- 

52 

53 def scores(self, date: str = "now") -> list[Game]: 

54 """Every game for ``date`` (``YYYY-MM-DD`` or ``now``), display-ordered.""" 

55 payload = self._get(f"/score/{date}") 

56 games = _parse_items(payload.get("games") or [], Game.from_api, "game") 

57 games.sort(key=Game.sort_key) 

58 return games 

59 

60 def schedule(self, team: str) -> list[Game]: 

61 """Every game on ``team``'s season schedule, in start order.""" 

62 payload = self._get(f"/club-schedule-season/{team.strip().upper()}/now") 

63 games = _parse_items(payload.get("games") or [], Game.from_api, "game") 

64 games.sort(key=lambda g: g.start_utc) 

65 return games 

66 

67 def situation(self, game_id: int) -> Situation | None: 

68 """Special-teams state for one live game; None at even strength.""" 

69 payload = self._get(f"/gamecenter/{game_id}/landing") 

70 try: 

71 return Situation.from_api(payload.get("situation")) 

72 except (AttributeError, KeyError, TypeError, ValueError) as exc: 

73 log.warning("Skipping malformed situation for game %s: %s", game_id, exc) 

74 return None 

75 

76 def standings(self, date: str = "now") -> list[StandingsRow]: 

77 """Every team's current standings line, unsorted across conferences. 

78 

79 During the off-season this endpoint keeps serving the just-finished 

80 season's final standings rather than an empty result (verified with 

81 a live call) -- callers that care about "has the current season 

82 actually started" need a signal beyond just "rows came back", e.g. 

83 the favourite's own ``games_played``. 

84 """ 

85 payload = self._get(f"/standings/{date}") 

86 return _parse_items(payload.get("standings") or [], StandingsRow.from_api, "standings row")

88 # -- plumbing --------------------------------------------------------

90 def _get(self, path: str) -> dict[str, Any]: 

91 url = f"{self.base_url}{path}" 

92 try: 

93 response = self.session.get(url, timeout=self.timeout) 

94 response.raise_for_status() 

95 payload = response.json() 

96 except requests.RequestException as exc: 

97 raise NHLApiError(f"GET {url} failed: {exc}") from exc 

98 except ValueError as exc: 

99 raise NHLApiError(f"GET {url} returned invalid JSON: {exc}") from exc 

100 if not isinstance(payload, dict): 

101 raise NHLApiError(f"GET {url} returned {type(payload).__name__}, expected object") 

102 return payload

105def _parse_items(raw_items: list[Any], parse: Callable[[Any], Any], label: str) -> list[Any]: 

106 """Parse each entry with ``parse``, skipping and logging one that's malformed. 

107 

108 One bad game/row must not abort the whole response -- see #59. 

109 """ 

110 items = [] 

111 for raw in raw_items: 

112 try: 

113 items.append(parse(raw)) 

114 except (AttributeError, KeyError, TypeError, ValueError) as exc: 

115 log.warning("Skipping malformed %s entry: %s", label, exc) 

116 return items

119def _build_session() -> requests.Session: 

120 session = requests.Session() 

121 session.headers.update({"User-Agent": USER_AGENT, "Accept": "application/json"}) 

122 retry = Retry( 

123 total=3, 

124 backoff_factor=0.5, 

125 status_forcelist=(429, 500, 502, 503, 504), 

126 allowed_methods=frozenset({"GET"}), 

127 raise_on_status=False, 

128 ) 

129 adapter = HTTPAdapter(max_retries=retry) 

130 session.mount("https://", adapter) 

131 session.mount("http://", adapter) 

132 return session
