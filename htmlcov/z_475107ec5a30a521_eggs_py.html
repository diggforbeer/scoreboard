1"""The Easter egg tour (#509): ``scoreboard --eggs [name]``, a hidden way to see them. 

2 

3The tree screen's quote, squirrel and fire, the Thanksgiving turkey's run (and the roast that 

4replaces him on the day itself), the blowout flash and April Fools' Admirals are all rare by 

5design: the tree ones only exist in December, the turkey in November, April Fools on one day, and 

6the blowout needs a five-goal loss. Waiting for one on a 

7real board is not a plan, so the flag plays each in turn on the real panel, in a loop (Ctrl-C to 

8stop), or just the one named. 

9 

10Like the eggs themselves this is kept out of ``--help``, the README, ``screens.md`` and demo 

11mode (owner's call); ``tests/test_eggs.py`` pins that. The holiday fly-bys are documented 

12features, not hidden, and are already in ``--demo``. 

13""" 

14 

15from __future__ import annotations 

16 

17from datetime import UTC, datetime 

18from typing import NamedTuple 

19 

20from . import blowout 

21from .app import Scene 

22from .demo import demo_steps 

23from .display.holiday import TREE_EGGS, TURKEY_RUN_SECONDS, nth_weekday 

24 

25#: The tour's items, in order. 

26NAMES = ("quote", "squirrel", "fire", "blowout", "april_fools", "turkey", "roast") 

27#: A calm moment of the plain screen before and after each, so the start and end are seen. 

28PAUSE_SECONDS = 2.0 

29#: Where the tree screen's date is held, so its countdown shows whatever the real date is. 

30TOUR_DAY = (12, 13) 

31#: The turkey's screen only exists in November: the tour holds the date here for it. 

32THANKSGIVING_DAY = (11, 19) 

33 

34 

35def roast_day() -> tuple[int, int]: 

36 """Thanksgiving Day proper, for the roast that replaces the turkey on it (#626): the 4th 

37 Thursday of this year's November, whatever the real date is.""" 

38 day = nth_weekday(datetime.now(UTC).year, 11, 3, 4) 

39 return day.month, day.day 

40 

41 

42#: How long each April Fools screen is shown relabelled. 

43PRANK_SECONDS = 4.0 

44 

45 

46class Phase(NamedTuple): 

47 """One stretch of the tour: a scene, for a time, optionally with an egg playing.""" 

48 

49 scene: Scene 

50 seconds: float 

51 #: A tree egg (quote, squirrel, fire) to play from the phase's start. 

52 egg: str | None = None 

53 #: Draw as April Fools' Day on a Preds board (NSH shown as MIL). 

54 prank: bool = False 

55 #: (month, day) to hold the clock at for this phase; default ``TOUR_DAY``. 

56 day: tuple[int, int] | None = None 

57 

58 

59def _demo_scene(name: str) -> Scene: 

60 steps = demo_steps("NSH", datetime(2026, 10, 14, tzinfo=UTC)) 

61 return next(s.scene for s in steps if s.name == name) 

62 

63 

64def seconds(name: str) -> float: 

65 """How long the egg itself plays (the calm moments around it not counted).""" 

66 return sum( 

67 p.seconds 

68 for p in plan(name) 

69 if p.egg or p.prank or p.scene.kind == "blowout" or name == "roast" 

70 ) 

71 

72 

73def plan(name: str) -> list[Phase]: 

74 """The tour of ``name``: a calm moment, the egg, a calm moment.""" 

75 tree = Scene("christmas_tree") 

76 if name in {kind for kind, _odds, _s in TREE_EGGS}: 

77 length = {kind: s for kind, _odds, s in TREE_EGGS}[name] 

78 return [ 

79 Phase(tree, PAUSE_SECONDS), 

80 Phase(tree, length, egg=name), 

81 Phase(tree, PAUSE_SECONDS), 

82 ] 

83 if name == "turkey": 

84 screen = Scene("thanksgiving") 

85 return [ 

86 Phase(screen, PAUSE_SECONDS, day=THANKSGIVING_DAY), 

87 Phase(screen, TURKEY_RUN_SECONDS, egg="turkey", day=THANKSGIVING_DAY), 

88 Phase(screen, PAUSE_SECONDS, day=THANKSGIVING_DAY), 

89 ] 

90 if name == "roast": 

91 screen = Scene("thanksgiving") 

92 return [Phase(screen, PAUSE_SECONDS * 3, day=roast_day())] 

93 if name == "blowout": 

94 final = next( 

95 s.scene.game 

96 for s in demo_steps("NSH", datetime(2026, 10, 14, tzinfo=UTC)) 

97 if s.scene.kind == "game" and s.scene.game and s.scene.game.is_final 

98 ) 

99 return [ 

100 Phase(tree, PAUSE_SECONDS), 

101 Phase(Scene("blowout", final), blowout.SECONDS), 

102 Phase(tree, PAUSE_SECONDS), 

103 ] 

104 if name == "april_fools": 

105 game, table = _demo_scene("game_live"), _demo_scene("standings") 

106 return [ 

107 Phase(game, PAUSE_SECONDS + 1), 

108 Phase(game, PRANK_SECONDS, prank=True), 

109 Phase(table, PRANK_SECONDS, prank=True), 

110 Phase(game, PAUSE_SECONDS), 

111 ] 

112 raise KeyError(name)
