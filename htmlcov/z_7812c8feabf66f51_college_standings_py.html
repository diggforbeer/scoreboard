1"""College conference tables (#553): the standings payload as one table per conference. 

2 

3``/apis/v2/sports/football/college-football/standings?group=80`` has a child per conference with 

4``standings.entries[]``. Three things in it are not what they look like (``football_api.md``): 

5 

6* each entry's ``stats`` is a flat list in which the **same stat name repeats once per split** 

7 (overall, Home, Away, vs. Division, vs. Conf.), so a stat is found by its ``type`` (``total`` is 

8 the overall record, ``vsconf`` the conference record, ``playoffseed`` the conference seed), never 

9 by ``name``; 

10* the **Sun Belt is nested**: its node has two divisions and no ``standings`` of its own, and the 

11 seeds run across both, so its teams are merged into one table; 

12* the entries are **not in seed order** (a team with a lower seed can be listed later), so the 

13 table is sorted by ``playoffseed``. 

14 

15The FBS Independents (Notre Dame and UConn) play no conference schedule: their conference record is 

16``0-0`` and is left out (``conf`` is empty). 

17"""

27#: How ESPN's conference names are shortened to fit beside a record (4 letters of room). 

28CONFERENCE_SHORT = { 

29 "big ten": "B1G", 

30 "big 12": "B12", 

31 "pac-12": "PAC", 

32 "fbs indep.": "IND", 

33 "american": "AAC", 

34 "sun belt": "SBC", 

35 "mountain west": "MWC", 

36 "conference usa": "CUSA", 

37 "mid-american": "MAC",

41def short_conference(name: str) -> str: 

42 """``Big Ten`` as ``B1G``: the known names by table, any other upper-cased and cut to 4.""" 

43 key = name.strip().lower() 

44 return CONFERENCE_SHORT.get(key, key.upper()[:4])

47@dataclass(frozen=True, slots=True) 

48class ConferenceRow:

52 overall: str 

53 #: The conference record, ``5-1``; empty for a team with no conference schedule.

84 def conference_of(self, abbrev: str) -> ConferenceTable | None: 

85 key = abbrev.strip().upper() 

86 return next((c for c in self.conferences if any(r.abbrev == key for r in c.rows)), None)

89def _stat(stats: list[Any], kind: str) -> dict[str, Any] | None: 

90 """The stat of ``type`` ``kind``: its first (and, by type, only) match.""" 

91 return next((s for s in stats if isinstance(s, dict) and s.get("type") == kind), None)

94def _text(stat: dict[str, Any] | None, *keys: str) -> str: 

95 for key in keys: 

96 value = (stat or {}).get(key) 

97 if isinstance(value, str) and value.strip() and value.strip() != "-": 

98 return value.strip() 

99 return ""

102def _row(entry: Any) -> ConferenceRow | None: 

103 try: 

104 team = entry["team"] 

105 team_id, abbrev = int(team["id"]), str(team["abbreviation"]).strip().upper() 

106 except (KeyError, TypeError, ValueError): 

107 return None 

108 if not abbrev: 

109 return None 

110 stats = entry.get("stats") 

111 stats = stats if isinstance(stats, list) else [] 

112 overall = _text(_stat(stats, "total"), "summary", "displayValue") 

113 conf = _text(_stat(stats, "vsconf"), "summary", "displayValue") 

114 if conf in ("0-0", "0-0-0"): 

115 conf = "" # no conference game played (an Independent, or the first week) 

116 try: 

117 seed: int | None = int(_text(_stat(stats, "playoffseed"), "displayValue")) 

118 except ValueError: 

119 seed = None 

120 if seed is not None and seed < 1: 

121 seed = None 

122 return ConferenceRow( 

123 team_id, abbrev, overall, conf, seed, _text(_stat(stats, "streak"), "displayValue") 

124 )

127def _entries(node: Any) -> list[Any]: 

128 """Every standings entry at or below ``node`` (the Sun Belt's are one level down).""" 

129 found: list[Any] = [] 

130 if not isinstance(node, dict): 

131 return found 

132 standings = node.get("standings") 

133 listed = standings.get("entries") if isinstance(standings, dict) else None 

134 if isinstance(listed, list): 

135 found.extend(listed) 

136 for child in node.get("children") or []: 

137 found.extend(_entries(child)) 

138 return found

141def _order(row: ConferenceRow) -> tuple[int, str]: 

142 """Seed order; a team without a seed goes last, in a stable order.""" 

143 return (row.seed if row.seed is not None else 10**6, row.abbrev)

146def college_standings_from_payload(payload: Any) -> CollegeStandings | None: 

147 """Every conference's table from a ``group=80`` standings payload, or ``None`` when it holds 

148 none (the wrong shape, or no entries). An entry that cannot be read is skipped.""" 

149 children = payload.get("children") if isinstance(payload, dict) else None 

150 tables: list[ConferenceTable] = [] 

151 for node in children if isinstance(children, list) else []: 

152 if not isinstance(node, dict): 

153 continue 

154 rows = [r for r in (_row(e) for e in _entries(node)) if r is not None] 

155 if not rows: 

156 continue 

157 name = str(node.get("name") or "").strip() 

158 short = short_conference(str(node.get("shortName") or name)) 

159 tables.append(ConferenceTable(name, short, tuple(sorted(rows, key=_order)))) 

160 return CollegeStandings(tuple(tables)) if tables else None
