1"""How far the scoreboard feed runs ahead of the play-by-play (a measurement, not a behaviour). 

2 

3The scoreboard names the play its live ``situation`` reflects (``lastPlay.id``); the plays feed 

4(``/summary``) lists plays by the same ids. The resting field, the score and the possession come 

5from the first, the animations from the second, so the field can show a play's result before the 

6play is shown. This logs, per watched game, when the scoreboard first names a play and when the 

7plays feed first has it:: 

8 

9 [401873077] scoreboard names play 4018730772112 (Punt); plays feed has it: no 

10 [401873077] plays feed has 4018730772112 (Punt) 11.2s after the scoreboard named it 

11 

12The board's own ``play shown: id=...`` line (``queue.py``) gives the third time, so the whole chain 

13is in ``journalctl -u scoreboard | grep -E "skew|play shown"``. Whether the ids really match in a 

14live game is part of what this answers: a play the scoreboard names that the feed never lists is 

15logged as ``never listed`` once the game moves on. 

16""" 

17 

18from __future__ import annotations 

19 

20import logging 

21import time 

22from collections.abc import Callable, Iterable 

23 

24from .models import FbGame, Play 

25 

26log = logging.getLogger(__name__) 

27 

28#: Plays remembered per game: enough for the lag, small enough to forget a finished game. 

29KEEP = 40 

30#: A named play still absent from the feed after this many newer plays were named is "never listed". 

31NEVER_AFTER = 3 

32 

33 

34class SkewLog: 

35 """Feed worker thread only.""" 

36 

37 def __init__(self, monotonic: Callable[[], float] = time.monotonic) -> None: 

38 self._monotonic = monotonic 

39 #: game id -> play id -> [monotonic time named, type, listed?, plays named since] 

40 self._named: dict[int, dict[str, list]] = {} 

41 

42 def observe(self, game: FbGame, plays: Iterable[Play]) -> None: 

43 """One poll: the game as the scoreboard has it, and its plays as the feed has them.""" 

44 now = self._monotonic() 

45 named = self._named.setdefault(game.id, {}) 

46 listed = {p.id for p in plays} 

47 last = game.situation.last_play if game.situation is not None else None 

48 if last is not None and last.id and last.id not in named: 

49 for entry in named.values(): 

50 entry[3] += 1 

51 named[last.id] = [now, last.type, False, 0] 

52 log.info( 

53 "[%s] scoreboard names play %s (%s); plays feed has it: %s", 

54 game.id, 

55 last.id, 

56 last.type, 

57 "yes" if last.id in listed else "no", 

58 ) 

59 for play_id, entry in list(named.items()): 

60 named_at, kind, was_listed, newer = entry 

61 if was_listed: 

62 continue 

63 if play_id in listed: 

64 entry[2] = True 

65 log.info( 

66 "[%s] plays feed has %s (%s) %.1fs after the scoreboard named it", 

67 game.id, 

68 play_id, 

69 kind, 

70 now - named_at, 

71 ) 

72 elif newer >= NEVER_AFTER: 

73 entry[2] = True 

74 log.info( 

75 "[%s] play %s (%s) named by the scoreboard, never listed", 

76 game.id, 

77 play_id, 

78 kind, 

79 ) 

80 while len(named) > KEEP: 

81 del named[next(iter(named))] 

82 

83 def forget(self, keep: Callable[[int], bool]) -> None: 

84 for game_id in [g for g in self._named if not keep(g)]: 

85 del self._named[game_id]
