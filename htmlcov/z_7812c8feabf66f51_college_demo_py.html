78def _game( 

79 away: str, 

80 home: str, 

81 state: str, 

82 a: int, 

83 h: int, 

84 short: str, 

85 period: int, 

86 ranks: tuple[int | None, int | None] = (None, None), 

87 situation: tuple[str, int, str, bool] | None = None, 

88 name: str = "STATUS_IN_PROGRESS", 

89 start: datetime | None = None, 

90 records: tuple[str, str] = ("", ""), 

91 gid: int = 1, 

92) -> FbGame: 

93 """A game as ESPN sends one, made in the demo's directory. ``situation`` is (possession, 

94 yard line from the home goal, down text, red zone); ``records`` are (away, home).""" 

95 

96 def side(abbrev: str, home_away: str, score: int, record: str) -> dict[str, Any]: 

97 out: dict[str, Any] = { 

98 "homeAway": home_away, 

99 "score": str(score), 

100 "team": {"id": str(_IDS[abbrev]), "abbreviation": abbrev}, 

101 } 

102 if record: 

103 out["records"] = [{"type": "total", "summary": record}] 

104 return out 

105 

106 when = (start or datetime(2026, 11, 21, 19, 30, tzinfo=UTC)).astimezone(UTC) 

107 stamp = when.strftime("%Y-%m-%dT%H:%MZ") 

108 comp: dict[str, Any] = { 

109 "id": str(gid), 

110 "startDate": stamp, 

111 "status": { 

112 "period": period, 

113 "displayClock": "3:12", 

114 "clock": 192.0, 

115 "type": { 

116 "state": state, 

117 "name": name, 

118 "shortDetail": short, 

119 "detail": short, 

120 "completed": state == "post", 

121 }, 

122 }, 

123 "competitors": [side(home, "home", h, records[1]), side(away, "away", a, records[0])], 

124 } 

125 if situation is not None: 

126 possession, yard, text, red = situation 

127 comp["situation"] = { 

128 "down": int(text[0]), 

129 "distance": 6, 

130 "yardLine": yard, 

131 "downDistanceText": f"{text} at X {yard}", 

132 "shortDownDistanceText": text, 

133 "possession": str(_IDS[possession]), 

134 "isRedZone": red, 

135 } 

136 with use_directory(DEMO_DIRECTORY): 

137 game = FbGame.from_api({"id": str(gid), "date": stamp, "competitions": [comp]}) 

138 return dataclasses.replace( 

139 game, 

140 away=dataclasses.replace(game.away, rank=ranks[0]), 

141 home=dataclasses.replace(game.home, rank=ranks[1]), 

142 )

145def _play(kind: str, text: str, team: str, score: tuple[int, int]) -> Play: 

146 pos = {

149 "down": 2, 

150 "distance": 7, 

151 "shortDownDistanceText": "2nd & 7", 

152 } 

153 return Play.from_api( 

154 { 

155 "id": f"college-{kind}", 

156 "sequenceNumber": "100", 

157 "type": {"text": kind}, 

158 "text": text, 

159 "period": {"number": 2}, 

160 "clock": {"displayValue": "6:41"}, 

161 "statYardage": 2, 

162 "scoringPlay": True, 

163 "isTurnover": False, 

164 "awayScore": score[0], 

165 "homeScore": score[1], 

166 "start": pos, 

167 "end": pos, 

168 } 

169 )

245 def upset(game: FbGame, final: bool, at: float) -> Scene: 

246 behind, ahead = (game.away, game.home) if game.away.score < game.home.score else ( 

247 game.home, game.away, 

248 ) # fmt: skip 

249 found = Upset(behind, ahead, final, ahead.score - behind.score) 

250 return Scene("fb_upset", fb_game=game, fb_upset=found, fb_elapsed=at)
