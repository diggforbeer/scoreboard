1"""``fb_score_detail`` (#391): who scored, after the celebration. 

2 

3Follows the touchdown or field-goal celebration for ``fb_score_detail_seconds`` (8s), like the 

4Hockey's ``goal_detail``. Between the two teams' logos (the scoring team's at full brightness, the 

5other dimmed) four short lines:: 

6 

7 14 C. WOERNER the scorer's jersey number and name (name alone with no number) 

8 1 YD PASS FROM the yards and how: PASS FROM / RUN / INT RETURN / FUM RETURN 

9 M. PENIX the passer (a pass only) 

10 PAT GOOD ATL 7 CIN 0 the extra point or two-point try, and the new score, scorer first 

11 

12After a field goal: the kicker (``14 N. FOLK``), ``31 YD FIELD GOAL``, and the new score. The score 

13is the one on the play itself, which already includes the point after. A single 64px panel has no 

14logos and drops the passer's line first, then the point-after text. Static: it does not move. 

15"""

39def _upper(name: str) -> str: 

40 return name.upper().strip()

43def person_line(person: Person | None, fallback: str) -> str: 

44 """``14 C. WOERNER``: the number then the name; the name alone with no number; the text's own 

45 spelling (``C.WOERNER``) when the roster does not know them; ``""`` when nobody is named.""" 

46 if person is not None: 

47 name = _upper(person.name or fallback) 

48 return f"{person.jersey} {name}".strip() if person.jersey else name 

49 return _upper(fallback)

52def scorer_name(scene: Scene) -> str: 

53 """The text's name for whoever carried or caught it: the receiver, rusher or returner.""" 

54 event, info = scene.fb_event, scene.fb_play.info 

55 if isinstance(event, Touchdown): 

56 if event.via == "pass": 

57 return info.target 

58 if event.via == "run": 

59 return info.rusher 

60 if event.via == "interception": 

61 return info.interceptor or info.returner 

62 return info.returner or info.rusher or info.target 

63 return info.kicker

66def scorer_line(scene: Scene) -> str: 

67 event = scene.fb_event 

68 roles = ("kicker",) if isinstance(event, FieldGoal) else ("scorer", "receiver", "rusher") 

69 return person_line(find(scene.fb_people, *roles), scorer_name(scene))

72def how_lines(scene: Scene) -> tuple[str, str]: 

73 """The yards and how, and (for a pass) the passer: ``("1 YD PASS FROM", "M. PENIX")``.""" 

74 event, play = scene.fb_event, scene.fb_play 

75 if isinstance(event, FieldGoal): 

76 yards = event.yards if event.yards is not None else play.info.fg_yards 

77 return (f"{yards} YD FIELD GOAL" if yards else "FIELD GOAL"), "" 

78 yards = f"{play.stat_yardage} YD " if play.stat_yardage else "" 

79 if event.via == "pass": 

80 passer = person_line(find(scene.fb_people, "passer"), play.info.passer) 

81 return f"{yards}PASS FROM", passer 

82 if event.via == "run": 

83 return f"{yards}RUN".strip(), "" 

84 if event.via == "interception": 

85 return f"{yards}INT RETURN".strip(), "" 

86 return f"{yards}FUM RETURN".strip() if play.info.fumble else f"{yards}RETURN".strip(), ""

89def point_after(scene: Scene) -> str: 

90 """``PAT GOOD``, ``PAT NO GOOD``, ``2 PT GOOD``, ``2 PT NO GOOD`` or ``""``.""" 

91 event = scene.fb_event 

92 if not isinstance(event, Touchdown): 

93 return "" 

94 if event.two_point: 

95 return {"good": "2 PT GOOD", "failed": "2 PT NO GOOD"}.get(event.two_point, "2 PT TRY") 

96 return {"good": "PAT GOOD", "no good": "PAT NO GOOD"}.get(event.extra_point, "")

99def score_line(scene: Scene) -> str: 

100 """``ATL 7 CIN 0``: the new score, the scoring team first.""" 

101 event, game = scene.fb_event, scene.fb_game 

102 away, home = (game.away.abbrev, event.away_score), (game.home.abbrev, event.home_score) 

103 mine, theirs = (home, away) if event.team == game.home.abbrev else (away, home) 

104 return f"{mine[0]} {mine[1]} {theirs[0]} {theirs[1]}"

107def other_team(scene: Scene) -> str: 

108 game = scene.fb_game 

109 return game.away.abbrev if scene.fb_event.team == game.home.abbrev else game.home.abbrev

112def _bottom(r: Renderer, canvas: Any, left: int, right: int, pat: str, score: str, passer: bool): 

113 """The point-after text at the left and the score at the right of the last row. When both do 

114 not fit: with no passer line the point-after text takes that row's place and the score stays 

115 below; with a passer it is dropped (the design's rule: the least important goes first).""" 

116 font = r.fonts.tiny 

117 gap = 4 

118 if pat and text_width(font, pat) + gap + text_width(font, score) <= right - left: 

119 r.text(canvas, font, left, 30, AMBER_CARD, pat) 

120 r.text_right(canvas, font, right, 30, AMBER_CARD, score) 

121 return 

122 if pat and not passer: 

123 r.text_center( 

124 canvas, font, (left + right) // 2, 22, AMBER_CARD, r_fit(pat, right - left, font) 

125 ) 

126 r.text_center( 

127 canvas, font, (left + right) // 2, 30, AMBER_CARD, r_fit(score, right - left, font) 

128 )

131def _logo(r: Renderer, canvas: Any, team: str, x: int, factor: float) -> None: 

132 logo = ( 

133 r.football_logos.get(team, LOGO_SIZE) 

134 if r.football_logos is not None and is_team(team) 

135 else None 

136 ) 

137 if logo is not None: 

138 ox, oy = x - logo.width // 2, 16 - logo.height // 2 

139 for dx, dy, (red, green, blue) in logo.pixels: 

140 canvas.SetPixel( 

141 ox + dx, oy + dy, round(red * factor), round(green * factor), round(blue * factor) 

142 ) 

143 return 

144 color = team_color(team) if is_team(team) else WHITE 

145 shade = tuple(round(c * factor) for c in color) 

146 r.text_center(canvas, r.fonts.small, x, 19, shade, r_fit(team, 22, r.fonts.small))

149@event_screen(Touchdown, DETAIL) 

150@event_screen(FieldGoal, DETAIL) 

151def draw_score_detail(r: Renderer, canvas: Any, scene: Scene) -> None: 

152 event = scene.fb_event 

153 canvas.Clear() 

154 wide = r.width >= 128 

155 color = team_color(event.team) if is_team(event.team) else WHITE 

156 left, right = (TEXT_LEFT, TEXT_RIGHT) if wide else (1, r.width - 1) 

157 cx, width = (left + right) // 2, right - left 

158 if wide: 

159 _logo(r, canvas, event.team, LOGO_X, 1.0) 

160 _logo(r, canvas, other_team(scene), r.width - LOGO_X, OTHER_TEAM_DIM) 

161 

162 how, passer = how_lines(scene) 

163 pat = point_after(scene) 

164 score = score_line(scene) 

165 r.text_center( 

166 canvas, r.fonts.small, cx, 8, WHITE, r_fit(scorer_line(scene), width, r.fonts.small) 

167 ) 

168 if wide: 

169 r.text_center(canvas, r.fonts.tiny, cx, 16, color, r_fit(how, width, r.fonts.tiny)) 

170 if passer: 

171 r.text_center(canvas, r.fonts.tiny, cx, 22, color, r_fit(passer, width, r.fonts.tiny)) 

172 _bottom(r, canvas, left, right, pat, score, bool(passer)) 

173 return 

174 # 64px: no passer line (it goes first); the point-after text stays, it still has a row. 

175 short_how = how.replace(" FROM", "") 

176 r.text_center( 

177 canvas, r.fonts.small, cx, 7, WHITE, r_fit(scorer_line(scene), width, r.fonts.small) 

178 ) 

179 r.text_center(canvas, r.fonts.tiny, cx, 14, color, r_fit(short_how, width, r.fonts.tiny)) 

180 if pat: 

181 r.text_center(canvas, r.fonts.tiny, cx, 21, AMBER_CARD, r_fit(pat, width, r.fonts.tiny)) 

182 r.text_center(canvas, r.fonts.tiny, cx, 28, AMBER_CARD, r_fit(score, width, r.fonts.tiny))
