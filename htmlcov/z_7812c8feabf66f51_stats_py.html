1"""Team stats for the break in play (the stats screen shown over an official timeout). 

2 

3Read from the **same** site ``summary`` payload the plays come from (``boxscore.teams[]``, each 

4with ``statistics[]`` of ``{name, label, displayValue}``), so showing them costs no request. The 

5screen is a split table, away on the left and home on the right, a handful of rows a page. 

6 

7Everything here is pure: :func:`stats_from_summary` parses, :class:`GameStats` answers "what does 

8each side say for this row, and who is ahead". 

9""" 

10 

11from __future__ import annotations 

12 

13import re 

14from dataclasses import dataclass 

15from typing import Any 

16 

17#: A row where the bigger number is better, the smaller is better, or there is no winner. 

18HIGHER, LOWER, NONE = "higher", "lower", "none" 

19#: Compared as a fraction (``4-5`` converted 4 of 5 -> 0.8), not as the first number. 

20RATIO = "ratio" 

21 

22 

23@dataclass(frozen=True, slots=True) 

24class StatRow: 

25 """One line of the table: the feed's key, the label drawn, and who is ahead.""" 

26 

27 key: str 

28 label: str 

29 better: str = HIGHER 

30 

31 

32#: The table, a page at a time (5 rows fit the 32px panel). Yardage first, as in the app the owner 

33#: showed; the rest are what a broadcast puts up at a break. 

34PAGES: tuple[tuple[StatRow, ...], ...] = ( 

35 ( 

36 StatRow("totalYards", "TOTAL YDS"), 

37 StatRow("netPassingYards", "PASSING"), 

38 StatRow("rushingYards", "RUSHING"), 

39 StatRow("yardsPerPlay", "YDS/PLAY"), 

40 StatRow("firstDowns", "1ST DOWNS"), 

41 ), 

42 ( 

43 StatRow("thirdDownEff", "3RD DOWN", RATIO), 

44 StatRow("turnovers", "TURNOVERS", LOWER), 

45 StatRow("totalPenaltiesYards", "PENALTIES", LOWER), 

46 StatRow("sacksYardsLost", "SACKS ALLOWED", LOWER), 

47 StatRow("possessionTime", "POSSESSION"), 

48 ), 

49) 

50

52#: Georgia at South Carolina, 2026-11-21), so the rows are the ones it does carry, 5 a page. 

53COLLEGE_PAGES: tuple[tuple[StatRow, ...], ...] = (

70# A minus only counts at the start of a number ("-2" yards), not as the dash in "4-5". 

71_NUMBER = re.compile(r"(?<!\d)-?\d+(?:\.\d+)?") 

72_COUNTS = re.compile(r"\d+") 

73 

74 

75@dataclass(frozen=True, slots=True) 

76class GameStats: 

77 """Both sides' feed values by key, as the feed displays them (``"197"``, ``"4-5"``).""" 

78 

79 away: dict[str, str] 

80 home: dict[str, str] 

81 #: The pages of rows to draw (the NFL's by default, ``COLLEGE_PAGES`` for a college game). 

82 layout: tuple[tuple[StatRow, ...], ...] = PAGES 

83 

84 def value(self, side: str, key: str) -> str: 

85 """The text for a side (``"away"`` / ``"home"``); ``"-"`` when the feed has none.""" 

86 return display((self.away if side == "away" else self.home).get(key, "")) 

87 

88 def leader(self, row: StatRow) -> str | None: 

89 """``"away"``, ``"home"``, or ``None`` (a tie, no winner, or an unreadable value).""" 

90 if row.better == NONE: 

91 return None 

92 a = _score(self.away.get(row.key, ""), row.better) 

93 h = _score(self.home.get(row.key, ""), row.better) 

94 if a is None or h is None or a == h: 

95 return None 

96 ahead = a > h 

97 if row.better == LOWER: 

98 ahead = not ahead 

99 return "away" if ahead else "home" 

100 

101 @property 

102 def pages(self) -> int: 

103 return len(self.layout) 

104 

105 def usable(self) -> bool: 

106 """True when the feed gave enough to be worth a screen (not a game with no snaps yet).""" 

107 first = self.layout[0][0].key 

108 return bool(self.away.get(first)) and bool(self.home.get(first)) 

109 

110 

111def display(raw: str) -> str: 

112 """The value as drawn: ``" 6:51"`` -> ``"6:51"``, ``"4-5"`` -> ``"4/5"``, ``""`` -> ``"-"``.""" 

113 text = raw.strip() 

114 if not text: 

115 return "-" 

116 if re.fullmatch(r"\d+-\d+", text): 

117 return text.replace("-", "/") 

118 return text 

119 

120 

121def _score(raw: str, better: str) -> float | None: 

122 """A comparable number for a value, or ``None`` if it has none.""" 

123 text = raw.strip() 

124 if not text: 

125 return None 

126 if ":" in text: # a possession clock, minutes:seconds 

127 minutes, _, seconds = text.partition(":") 

128 try: 

129 return int(minutes) * 60 + int(seconds) 

130 except ValueError: 

131 return None 

132 if better == RATIO: # "4-5": made and attempted 

133 counts = _COUNTS.findall(text) 

134 if not counts: 

135 return None 

136 attempts = float(counts[1]) if len(counts) > 1 else 0.0 

137 return float(counts[0]) / attempts if attempts else 0.0 

138 nums = _NUMBER.findall(text) 

139 return float(nums[0]) if nums else None

142def stats_from_summary( 

143 payload: dict[str, Any], layout: tuple[tuple[StatRow, ...], ...] = PAGES 

144) -> GameStats | None: 

145 """The team stats in a site ``summary`` payload, or ``None`` when it has none. 

146 

147 Sides are matched by ``homeAway``, never by order. A key the feed lists twice (it does for 

148 ``interceptions``) keeps its first value. Anything malformed is skipped, never raised. 

149 """ 

150 box = payload.get("boxscore") 

151 teams = box.get("teams") if isinstance(box, dict) else None 

152 if not isinstance(teams, list): 

153 return None 

154 sides: dict[str, dict[str, str]] = {} 

155 for team in teams: 

156 if not isinstance(team, dict) or team.get("homeAway") not in ("home", "away"): 

157 continue 

158 values: dict[str, str] = {} 

159 for item in team.get("statistics") or []: 

160 if not isinstance(item, dict): 

161 continue 

162 name, shown = item.get("name"), item.get("displayValue") 

163 if isinstance(name, str) and isinstance(shown, str): 

164 values.setdefault(name, shown) 

165 sides[team["homeAway"]] = values 

166 if "home" not in sides or "away" not in sides: 

167 return None 

168 stats = GameStats(away=sides["away"], home=sides["home"], layout=layout) 

169 return stats if stats.usable() else None
