54class LogoLibrary: 

55 """Finds and caches logos; a missing logo is ``None``, never an error."""

57 def __init__(self, directories: Sequence[Path], size: int = 32) -> None: 

58 self.directories = [Path(d) for d in directories] 

59 self.size = size 

60 self._cache: dict[str, Logo | None] = {} 

61 self._warned_no_pillow = False

63 @classmethod 

64 def default(cls, size: int = 32, variant: str = "dark") -> LogoLibrary:

67 def path_for(self, abbrev: str) -> Path | None: 

68 name = f"{abbrev.strip().upper()}.png" 

69 for directory in self.directories: 

70 candidate = directory / name

72 return candidate

76 key = abbrev.strip().upper() 

77 if key not in self._cache:

84 def _load(self, abbrev: str) -> Logo | None: 

85 path = self.path_for(abbrev) 

86 if path is None: 

87 log.debug("No logo for %s in %s", abbrev, self.directories) 

88 return None 

89 try: 

90 from PIL import Image 

91 except ImportError: 

92 if not self._warned_no_pillow: 

93 log.warning("Pillow not installed; logos disabled") 

94 self._warned_no_pillow = True 

95 return None 

96 try: 

97 with Image.open(path) as im: 

98 logo = decode(abbrev, im.convert("RGBA"), self.size) 

99 except OSError as exc: 

100 log.warning("Could not read logo %s: %s", path, exc) 

101 return None 

102 log.debug("Loaded logo %s (%d px lit)", path, len(logo.pixels)) 

103 return logo

106def decode(abbrev: str, image, size: int) -> Logo: 

107 """Flatten an RGBA image into lit pixels, composited over black.""" 

108 if image.size != (size, size): 

109 # Lazy, like _load's: Pillow is optional. Callers only get here with an 

110 # image Pillow already decoded, so no ImportError guard is needed. 

111 # LANCZOS matches fetch-logos.py; the default filter mushes fine 

112 # linework at 32px, which is the whole point of the override slot. 

113 from PIL import Image 

114 

115 image = image.resize((size, size), resample=Image.LANCZOS) 

116 pixels: list[tuple[int, int, RGB]] = [] 

117 data = image.load() 

118 for y in range(size): 

119 for x in range(size): 

120 r, g, b, a = data[x, y] 

121 if a < ALPHA_FLOOR: 

122 continue 

123 rgb = (r * a // 255, g * a // 255, b * a // 255) 

124 if max(rgb) < LUMA_FLOOR: 

125 continue 

126 pixels.append((x, y, rgb)) 

127 return Logo(abbrev=abbrev, width=size, height=size, pixels=tuple(pixels))
