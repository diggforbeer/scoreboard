1"""Typed events from plays and scores: what the screens will react to (#356). 

2 

3No drawing and no sound here, only the decision "something worth a screen just happened, 

4to whom, and how big". The screens (celebrations, the turnover alert, the field goal 

5screens) are later stories; they read these. 

6 

7Rules, from ``football_screen.md``: 

8 

9* **Who scored comes from the score delta**, never from possession, so a pick-six counts 

10 for the defence. The feed's own running score is on every play. 

11* **Scoping** decides how big, or whether, an event is: the favourite's scores are 

12 ``FULL`` (and carry the sound), a followed team's are ``QUICK``, anything else is not 

13 emitted at all. As on the NHL side, in the **favourite's own game only the favourite's 

14 scores count**; two followed teams meeting both count. The one exception is **field 

15 goals**: the outcome, made or missed, is shown for *either* team in a followed game 

16 (``PLAIN`` for a team that is not followed). Turnovers and the coin toss are about the 

17 game rather than a team: ``FULL`` in the favourite's game, ``QUICK`` in a followed game. 

18* **Nothing fires for the score the board started with.** A game's score is seeded from the 

19 play queue's baseline, and a fresh detector (a new mode after a sport switch) starts 

20 empty; an unseeded game treats its first play as the baseline rather than guess. 

21* **Skipped plays still count.** The queue may skip plays to catch up; a touchdown among 

22 them must still celebrate, so the detector reads every skipped play as well as the one 

23 shown, in order. 

24* **Sound** is a flag, not a sound: only the favourite's touchdown and made field goal. 

25"""

153 def retract(self, game_id: int, away_points: int, home_points: int) -> None: 

154 """Take points back off a game's running score after a reversal (#357). 

155 

156 A reversed touchdown leaves the tracked score too high: the next play's score comes 

157 from a game in which it never happened, so without this its real touchdown would show 

158 no change and go uncelebrated. Subtracting only the reversed points (not resetting to 

159 the rewritten play's score) keeps later, still-valid scores intact. 

160 """ 

161 away, home = self._scores.get(game_id, (0, 0)) 

162 if game_id in self._scores: 

163 self._scores[game_id] = (max(0, away - away_points), max(0, home - home_points))

167 for game_id in [g for g in self._scores if not keep(g)]: 

168 del self._scores[game_id]

184 def _on_play(self, game: FbGame, play: Play) -> list[Event]: 

185 cfg = self._football() 

186 before = self._scores.get(game.id) 

187 self._scores[game.id] = (play.away_score, play.home_score) 

188 out: list[Event] = [] 

189 

190 toss = self._coin_toss(game, play, cfg) 

191 if toss is not None: 

192 out.append(toss) 

193 if before is None: 

194 # Nothing seeded this game: this play is the baseline, not news. 

195 log.info("[%s] unseeded game: treating play %s as the baseline", game.id, play.id) 

196 return out 

197 

198 deltas = (play.away_score - before[0], play.home_score - before[1]) 

199 scored = False 

200 for side, delta in zip((game.away, game.home), deltas, strict=True): 

201 if delta > 0 and not play.scoring_play: 

202 # The feed publishes a touchdown, then rewrites it with the extra point and the 

203 # new total; the next play (the kickoff) already carries that total. The running 

204 # score has just caught up with a score that was announced with its touchdown: 

205 # not news, and celebrating it would be undone at once as a play that does not 

206 # score (a "SCORE OVERTURNED" after every touchdown). 

207 log.info( 

208 "[%s] score moved to %s-%s on %s, which is not a scoring play: not a score", 

209 game.id, play.away_score, play.home_score, play.id, 

210 ) # fmt: skip 

211 continue 

212 if delta > 0: 

213 scored = True 

214 event = self._score(game, play, side.abbrev, delta, before, cfg) 

215 if event is not None: 

216 out.append(event) 

217 

218 if not scored: 

219 miss = self._field_goal_miss(game, play, before, cfg) 

220 if miss is not None: 

221 out.append(miss) 

222 turnover = self._turnover(game, play, cfg)

318 def _coin_toss(self, game: FbGame, play: Play, cfg) -> CoinToss | None: 

319 """The receiver of the opening kickoff. 

320 

321 The site feed omits the ``Coin Toss`` play (``football_api.md``), so the opening 

322 kickoff names the receiver; a ``Coin Toss`` play, if one is ever given, does too. 

323 The opening kickoff is the one at 15:00 in the first quarter (a kickoff after an 

324 early score shows the time left), so a board that joins mid-game, whose baseline 

325 already holds it, never sees it again. 

326 """ 

327 if game.id in self._toss_done: 

328 return None 

329 opening_kick = play.kind == PlayKind.KICKOFF and play.period == 1 and play.clock == "15:00" 

330 if play.kind != PlayKind.COIN_TOSS and not opening_kick: 

331 return None 

332 receiver = _abbrev(game, play.end.team_id if play.end else None) 

333 kicker = _abbrev(game, play.start.team_id if play.start else None) 

334 self._toss_done.add(game.id) 

335 scope = _game_scope(game, cfg) 

336 if not receiver or scope is None: 

337 return None 

338 return CoinToss( 

339 play=play, game_id=game.id, team=receiver, scope=scope, kicker=kicker,

363def _score_scope(team: str, game: FbGame, cfg: FootballConfig) -> Scope | None: 

364 """A touchdown or other score: the favourite's is FULL, a followed team's QUICK. 

365 

366 In the favourite's own game only the favourite's scores count (a followed team scoring 

367 *on* the favourite gets no screen), as on the NHL side; two followed teams meeting both 

368 count; an opponent's score is not emitted. 

369 """ 

370 favourite, followed = _followed(cfg) 

371 if favourite and team == favourite: 

372 return Scope.FULL 

373 if team in followed: 

374 return None if favourite and game.involves(favourite) else Scope.QUICK 

375 return None
