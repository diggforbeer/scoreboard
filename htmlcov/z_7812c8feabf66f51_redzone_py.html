1"""The red zone alert: a team has driven inside the opponent's 20 (#redzone). 

2 

3Followed teams have their own screens, and the favourite's game takes the whole panel. This is 

4for the board when it is showing an *idle* screen (the countdown, the final, the standings, the 

5week, the schedule, the clock): a live game somewhere moves into scoring range, which is the 

6moment worth leaving the idle screen for. It is the league-wide "RedZone" idea, in one short card. 

7 

8It reads the scoreboard feed's ``situation`` (``isRedZone`` and who has the ball), so it costs no 

9extra requests, and it only *describes* the alert: whether the board may show it (the setting, a 

10128px panel, an idle screen up) is the mode's decision. 

11 

12Rules: 

13 

14* **A transition, not a state.** An alert is made when a game goes from out of the red zone to 

15 in it (or to the other team's red zone). A game that is already there when the board first 

16 sees it is only noted: nothing fires for what the board started with. 

17* **One per visit, spaced out.** A game that flickers in and out at the edge of the zone alerts 

18 once per ``COOLDOWN`` seconds. Several games entering together queue up and are shown one after 

19 another; one that has waited more than ``STALE_AFTER`` seconds is dropped, since by then it is 

20 old news. 

21* Only live games with a ball in play count: not the half, not a final. 

22""" 

23

26import logging 

27from collections import deque 

28from collections.abc import Callable, Sequence

33log = logging.getLogger(__name__)

35#: Seconds before the same game may alert again. 

36COOLDOWN = 120.0 

37#: Seconds a queued alert may wait before it is dropped. 

38STALE_AFTER = 30.0 

39 

40 

41@dataclass(frozen=True, slots=True) 

42class RedZoneAlert: 

43 game_id: int 

44 #: ESPN team id of the team with the ball. 

45 team_id: int 

46 queued_at: float

49def attacking_team(game: FbGame) -> int | None: 

50 """The ESPN team id of the team in the red zone, or ``None`` if the game is not in one.""" 

51 if not game.is_live or game.is_halftime or game.situation is None: 

52 return None 

53 situation = game.situation 

54 if not situation.is_red_zone or situation.possession_team_id is None: 

55 return None 

56 if situation.possession_team_id not in (game.away.espn_id, game.home.espn_id): 

57 return None 

58 return situation.possession_team_id

61class RedZoneWatcher: 

62 def __init__(self, monotonic: Callable[[], float]) -> None: 

63 self._monotonic = monotonic 

64 #: game id -> the team id that was in the zone at the last look (``None``: nobody) 

65 self._zone: dict[int, int | None] = {} 

66 self._last_alert: dict[int, float] = {} 

67 self._pending: deque[RedZoneAlert] = deque() 

68 self._active: tuple[RedZoneAlert, float] | None = None 

69 

70 def observe(self, games: Sequence[FbGame]) -> None: 

71 """Look at the slate as the feed last gave it. Cheap: call it as often as you like.""" 

72 now = self._monotonic() 

73 seen = set() 

74 for game in games: 

75 seen.add(game.id) 

76 team = attacking_team(game) 

77 known = game.id in self._zone 

78 before = self._zone.get(game.id) 

79 self._zone[game.id] = team 

80 if team is None or not known or team == before: 

81 continue # not in the zone, first sight (noted only), or still in the same one 

82 if now - self._last_alert.get(game.id, -COOLDOWN) < COOLDOWN: 

83 continue 

84 self._last_alert[game.id] = now 

85 self._pending.append(RedZoneAlert(game.id, team, now)) 

86 log.info("[%s] red zone: team %s has driven inside the 20", game.id, team) 

87 for gone in [g for g in self._zone if g not in seen]: 

88 del self._zone[gone] 

89 self._last_alert.pop(gone, None) 

90 

91 def current(self, games: Sequence[FbGame], seconds: float) -> tuple[FbGame, float] | None: 

92 """The alert on screen as ``(game, seconds in)``, starting the next queued one when the 

93 last has had its ``seconds``; ``None`` when there is nothing to show.""" 

94 now = self._monotonic() 

95 if self._active is not None and now - self._active[1] >= seconds: 

96 self._active = None 

97 by_id = {g.id: g for g in games} 

98 while self._active is None and self._pending: 

99 alert = self._pending.popleft() 

100 if now - alert.queued_at > STALE_AFTER or alert.game_id not in by_id: 

101 continue 

102 self._active = (alert, now) 

103 if self._active is None: 

104 return None 

105 alert, started = self._active 

106 game = by_id.get(alert.game_id) 

107 if game is None: 

108 self._active = None 

109 return None 

110 return game, now - started 

111 

112 @property 

113 def waiting(self) -> int: 

114 return len(self._pending) 

115 

116 def clear(self) -> None: 

117 self._zone.clear() 

118 self._last_alert.clear() 

119 self._pending.clear() 

120 self._active = None
