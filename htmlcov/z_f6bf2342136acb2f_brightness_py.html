1"""Lux -> LED brightness mapping for auto-dimming (#44). 

2 

3Both human brightness perception and a cheap ambient sensor's usable range 

4are roughly logarithmic, not linear: the difference between 1 lux and 10 lux 

5(a dark room vs. a dim lamp) should move the panel about as much as 10 lux 

6to 100 lux (a dim lamp vs. normal room light). A linear map spends nearly 

7its whole range on the first few lux and then goes flat for the rest of a 

8typical room's brightness swing. 

9""" 

10 

11from __future__ import annotations 

12 

13import math 

14 

15#: Lux at and beyond which the panel is already at max_brightness. This is 

16#: "a brightly lit room", not direct sunlight -- a living-room display never 

17#: needs to out-shine a window, and this board has no sensor housing rated 

18#: for outdoor use anyway. 

19REFERENCE_LUX = 1000.0 

20 

21 

22def lux_to_brightness(lux: float, min_brightness: int, max_brightness: int) -> int: 

23 """Map a lux reading onto ``[min_brightness, max_brightness]``.""" 

24 lux = max(lux, 0.0) 

25 span = max_brightness - min_brightness 

26 fraction = math.log1p(lux) / math.log1p(REFERENCE_LUX) 

27 fraction = min(max(fraction, 0.0), 1.0) 

28 return round(min_brightness + span * fraction)
