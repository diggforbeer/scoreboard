63def build_poll(ctx: IdleContext, key: str, cut: int = 0) -> PollView | None: 

64 """The poll ``key`` as a view, or ``None`` when there is no fresh copy of it: not fetched yet, 

65 or older than ``FRESH_DAYS`` (out of season), so the screen is skipped rather than shown blank. 

66 ``chosen()`` is not used: the AP's screen is the AP's whatever ``college_poll`` says.""" 

67 polls = ctx.idle.polls() 

68 poll = polls.get(key, ctx.utc) if polls is not None else None 

69 if poll is None or not poll.entries: 

70 return None 

71 rows = tuple( 

72 PollRow( 

73 rank=e.rank, 

74 abbrev=e.abbrev, 

75 record=e.record, 

76 movement=e.movement, 

77 points=int(e.points), 

78 firsts=e.first_place_votes, 

79 conference=short_conference(e.conference), 

80 ) 

81 for e in poll.entries 

82 ) 

83 return PollView( 

84 key=key, 

85 title=POLL_NAMES[key].upper(), 

86 label=POLL_LABELS[key], 

87 rows=rows, 

88 team=ctx.config.favourite_team.strip().upper(), 

89 cut=cut, 

90 )

94 return build_poll(ctx, POLL_AP)

97def build_cfp(ctx: IdleContext) -> PollView | None: 

98 """The Playoff Committee's rankings (#552), only once they exist for this season: they start 

99 around week 10, so before that ESPN sends none (``None``), and last season's last table is

121def build_conference(ctx: IdleContext) -> ConferenceView | None: 

122 """The lead team's conference, or ``None`` (skip, never blank) when the standings are not 

123 fetched yet or stale, the team is in none, or no team in it has played a game (the table is 

124 all zeros before the season).""" 

125 standings = ctx.idle.college_standings(ctx.now) 

126 team = ctx.config.favourite_team.strip().upper() 

127 table = standings.conference_of(team) if standings is not None and team else None 

128 if table is None or not table.played: 

129 return None 

130 return ConferenceView(table.name, table.short, table.rows, team)

133# -- the scores grid (#556) ----------------------------------------------------------------------- 

134 

135#: Games on one frame of the grid: four 32px columns across the 128px chain (idle screens are 128px 

136#: only), the same as hockey's ``scores_grid``. 

137SCORES_PER_PAGE = 4

140@dataclass(frozen=True, slots=True) 

141class ScoresView: 

142 """The slate for the scores grid, ranked games first, in pages of ``SCORES_PER_PAGE``.""" 

143 

144 games: tuple[FbGame, ...] 

145 

146 @property 

147 def pages(self) -> int: 

148 return max(-(-len(self.games) // SCORES_PER_PAGE), 1) 

149 

150 def page(self, number: int) -> tuple[FbGame, ...]: 

151 start = number * SCORES_PER_PAGE 

152 return self.games[start : start + SCORES_PER_PAGE]

155def _best_rank(game: FbGame) -> int: 

156 ranks = [t.rank for t in (game.away, game.home) if t.rank] 

157 return min(ranks) if ranks else 99 

158 

159 

160def scores_order(games: Sequence[FbGame], ranked_only: bool) -> tuple[FbGame, ...]: 

161 """Games with a ranked team first (best rank leading), then the rest; each group in kickoff 

162 order, so the order is steady while a game's score changes. ``ranked_only`` drops the rest.""" 

163 far = datetime.max.replace(tzinfo=UTC) 

164 picked = [g for g in games if _best_rank(g) < 99 or not ranked_only] 

165 return tuple(sorted(picked, key=lambda g: (_best_rank(g), g.start_utc or far, g.id)))

168def build_scores(ctx: IdleContext) -> ScoresView | None: 

169 """The scoreboard already polled, filtered by ``football.college_scores``; ``None`` with 

170 nothing to list (the slot is skipped, never shown blank).""" 

171 games = scores_order(ctx.games, getattr(ctx.config, "college_scores", "all") == "ranked") 

172 return ScoresView(games) if games else None
