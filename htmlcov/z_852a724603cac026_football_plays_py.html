1"""The football play scenes (#377): what is drawn while a play, its text or an event is on screen. 

2 

3Three scene kinds, chosen by ``football.director``: 

4 

5* ``fb_play``: the animation of one play, a pure function of the seconds since it started. 

6* ``fb_playtext``: the play's own sentence once the animation is over. 

7* ``fb_event``: a celebration, alert, coin toss or reversal (``Scene.fb_stage`` says which). 

8 

9Every designed screen is **one entry in a registry**, so a screen's story only adds a function: 

10 

11* ``@play_screen(PlayKind.RUSH)`` registers the animation for a play kind; 

12* ``@event_screen(Touchdown, EVENT)`` registers the screen for an event class in a stage 

13 (``football.director.EVENT`` / ``DETAIL`` / ``REVERSAL``). 

14 

15Anything with no entry gets the generic version here, so an unknown play type or an event nobody 

16has drawn yet still shows something sensible and never raises. These generic screens are 

17deliberately plain: they are the floor every designed screen replaces. 

18"""

98def load_screens() -> None: 

99 """Import every screen module so its ``@play_screen`` / ``@event_screen`` entries exist.""" 

100 global _loaded 

101 if _loaded: 

102 return 

103 _loaded = True 

104 for name in SCREEN_MODULES: 

105 importlib.import_module(f".{name}", __package__)

108def draw(r: Renderer, canvas: Any, scene: Scene) -> None: 

109 """Draw one of the three play scenes.""" 

110 load_screens() 

111 if scene.kind == "fb_play": 

112 PLAY_SCREENS.get(scene.fb_play.kind, draw_generic_play)(r, canvas, scene) 

113 elif scene.kind == "fb_playtext": 

114 draw_playtext(r, canvas, scene) 

115 else: 

116 drawer = EVENT_SCREENS.get((type(scene.fb_event), scene.fb_stage), draw_generic_event) 

117 drawer(r, canvas, scene)

123def play_label(scene: Scene) -> tuple[str, tuple[int, int, int]]: 

124 """``PUNT +53``-style text for the band and its colour: green for a gain, red for a loss.""" 

125 play = scene.fb_play 

126 yards = play.stat_yardage 

127 name = play.type.upper() or play.kind.value.replace("_", " ").upper() 

128 if play.kind in (PlayKind.RUSH, PlayKind.PASS_COMPLETE, PlayKind.SACK) and yards: 

129 return f"{name.split()[0]} {yards:+d}", GAIN if yards > 0 else LOSS 

130 return name, NEUTRAL

133def quarter_clock(play: Any) -> str: 

134 """``3RD 12:24`` for the band (the 4x6 ``Q`` reads as a zero); ``OT`` after the fourth.""" 

135 ordinal = {1: "1ST", 2: "2ND", 3: "3RD", 4: "4TH"}.get(play.period, "OT") 

136 return f"{ordinal} {play.clock}".strip()

147def draw_generic_play(r: Renderer, canvas: Any, scene: Scene) -> None: 

148 """Any play: the field, and the ball sliding from where the play started to where it ended, 

149 with the play's name over it.""" 

150 game, play = scene.fb_game, scene.fb_play 

151 canvas.Clear() 

152 layout = FieldLayout.for_panel(r.width) 

153 team_id = (play.start or play.end).team_id if (play.start or play.end) else None 

154 side = _side(game, team_id) 

155 ff.draw_field( 

156 r, canvas, layout, game.home.abbrev, game.away.abbrev, side, logos=r.football_logos 

157 ) 

158 start = _spot(play.start, 50.0) 

159 end = _spot(play.end, start) 

160 progress = ( 

161 1.0 if scene.fb_seconds <= 0 else min(1.0, scene.fb_elapsed / (scene.fb_seconds * TRAVEL)) 

162 ) 

163 ff.draw_ball(canvas, layout, start + (end - start) * progress) 

164 label, color = play_label(scene) 

165 _band(r, canvas, label, color, quarter_clock(play))

168def _band(r: Renderer, canvas: Any, label: str, color: tuple[int, int, int], right: str) -> None: 

169 """The one-line band over the field: the play on the left, the clock on the right.""" 

170 clock = r._fit_text(r.fonts.tiny, right.upper(), r.width // 2 - 2) 

171 r.text_right(canvas, r.fonts.tiny, r.width - 1, 6, DIM, clock) 

172 room = r.width - text_width(r.fonts.tiny, clock) - 4 

173 r.text(canvas, r.fonts.tiny, 1, 6, color, r._fit_text(r.fonts.tiny, label, room))

179def wrap(text: str, width: int, font: Any, lines: int) -> list[str]: 

180 """``text`` broken into at most ``lines`` lines of ``width`` pixels (the last cut short).""" 

181 out: list[str] = [] 

182 line = "" 

183 for word in text.split(): 

184 trial = f"{line} {word}".strip() 

185 if text_width(font, trial) <= width: 

186 line = trial 

187 continue 

188 if line: 

189 out.append(line) 

190 line = word 

191 if line: 

192 out.append(line) 

193 if len(out) > lines: 

194 out = [*out[: lines - 1], r_fit(" ".join(out[lines - 1 :]), width, font)] 

195 return [r_fit(x, width, font) for x in out]

204def draw_playtext(r: Renderer, canvas: Any, scene: Scene) -> None: 

205 """The field and the feed's own sentence (#379). 

206 

207 128px: the field in the left half with the ball where the play left it, the sentence on the 

208 right three lines at a time (it scrolls a line at a time when longer), and the new down and 

209 distance under it. 64px: the field and the text take turns (``football.textlayout``). 

210 """ 

211 play = scene.fb_play 

212 canvas.Clear() 

213 wide = r.width >= 128 

214 pos = play.end or play.start 

215 down = pos.short_down_distance_text.upper() if pos else "" 

216 if wide: 

217 half = r.width // 2 

218 _field_half(r, canvas, scene, 0, half) 

219 x = half + 3 

220 lines = wrap(play.text.upper(), r.width - x - 1, r.fonts.tiny, 99) 

221 _text_lines(r, canvas, lines, x, tl.WIDE_LINES, scene.fb_elapsed, down) 

222 return 

223 visible = NARROW_LINES 

224 lines = wrap(play.text.upper(), r.width - 3, r.fonts.tiny, 99) 

225 view, t = tl.narrow_view(scene.fb_elapsed, len(lines), visible) 

226 if view == "field": 

227 _field_half(r, canvas, scene, 0, r.width) 

228 else: 

229 _text_lines(r, canvas, lines, 2, visible, t, down)

233NARROW_LINES = 3

236def _field_half(r: Renderer, canvas: Any, scene: Scene, left: int, right: int) -> None: 

237 """The field in ``left..right`` with the ball where the play left it, and its label.""" 

238 game, play = scene.fb_game, scene.fb_play 

239 layout = FieldLayout.for_panel(r.width, left, right) 

240 pos = play.end or play.start 

241 side = _side(game, pos.team_id if pos else None) 

242 ff.draw_field(r, canvas, layout, game.home.abbrev, game.away.abbrev, side, logos=None) 

243 if pos is not None and pos.yard_line is not None: 

244 ff.draw_ball(canvas, layout, pos.yard_line) 

245 label, color = play_label(scene) 

246 r.text(canvas, r.fonts.tiny, left + 1, 6, color, r_fit(label, right - left - 2, r.fonts.tiny))

249def _text_lines( 

250 r: Renderer, canvas: Any, lines: list[str], x: int, visible: int, elapsed: float, down: str 

251) -> None: 

252 """``visible`` lines of text from where the scroll has got to, with the down under them.""" 

253 first = tl.first_line(elapsed, len(lines), visible) 

254 for i, line in enumerate(lines[first : first + visible]): 

255 r.text(canvas, r.fonts.tiny, x, 6 + 7 * i, WHITE, line) 

256 if down: 

257 r.text( 

258 canvas, 

259 r.fonts.tiny, 

260 x, 

261 r.height - 2, 

262 NEUTRAL, 

263 r_fit(down, r.width - x - 1, r.fonts.tiny), 

264 )

270def event_title(scene: Scene) -> tuple[str, tuple[int, int, int]]: 

271 """The headline and colour of an event: its team's colour for a score, green or red for a 

272 turnover by who gained the ball, amber for a correction or a plain kick.""" 

273 event = scene.fb_event 

274 team = event.team 

275 team_rgb = team_color(team) if is_team(team) else WHITE 

276 if isinstance(event, Reversal): 

277 return event.title, NEUTRAL 

278 if isinstance(event, Touchdown): 

279 return "TOUCHDOWN", team_rgb 

280 if isinstance(event, FieldGoal): 

281 return ("FIELD GOAL" if event.made else "NO GOOD"), (team_rgb if event.made else NEUTRAL) 

282 if isinstance(event, GenericScore): 

283 return f"{event.points} POINTS", team_rgb 

284 if isinstance(event, Turnover): 

285 return event.reason.replace("_", " ").upper() or "TURNOVER", NEUTRAL 

286 if isinstance(event, CoinToss): 

287 return f"{team} RECEIVE", team_rgb 

288 return "PLAY", WHITE

296def draw_generic_event(r: Renderer, canvas: Any, scene: Scene) -> None: 

297 """A full-panel card: the headline in colour, and under it the detail or the new score.""" 

298 canvas.Clear() 

299 title, color = event_title(scene) 

300 width = r.width - 4 

301 font = next( 

302 ( 

303 f 

304 for f in (r.fonts.large, r.fonts.medium, r.fonts.small) 

305 if text_width(f, title) <= width 

306 ), 

307 r.fonts.tiny, 

308 ) 

309 r.text_center(canvas, font, r.width // 2, 13, color, r_fit(title, width, font)) 

310 if scene.fb_stage == DETAIL: 

311 line = scene.fb_play.text.upper() 

312 elif scene.fb_stage == REVERSAL: 

313 line = getattr(scene.fb_event, "reason", "").upper() 

314 elif isinstance(scene.fb_event, CoinToss): 

315 line = "KICKS " + scene.fb_event.kicker if scene.fb_event.kicker else "" 

316 else: 

317 line = _score_line(scene) 

318 r.text_center(canvas, r.fonts.tiny, r.width // 2, 22, WHITE, r_fit(line, width, r.fonts.tiny))
