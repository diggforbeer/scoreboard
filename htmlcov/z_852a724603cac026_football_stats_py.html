1"""The split team stats shown over an official timeout or the two-minute warning. 

2 

3A TV timeout is the one stretch where nothing is moving, so the board shows how the game is going 

4instead of a field with a ball on it. Away on the left, home on the right (like the text 

5scoreboard), each team's 16px logo at its edge, and between them a table: the stat's name in the 

6middle, each side's number beside its logo. Five rows a page, the pages sharing the stage's 

7seconds (``stats_seconds``). Whoever is ahead on a row shows it in their team colour and the other 

8side dims, so the leader reads at a glance; a tie keeps both lit. Under each logo a pair of dots 

9says which page this is. 

10 

11A pure function of the seconds since the stage began, like every other play screen. 128px only: 

12the director never starts the stage on a narrower panel (a split table does not fit 64px). 

13"""

29ROW_BASELINES = (6, 12, 18, 24, 30)

38def page_of(scene: Scene, pages: int) -> int: 

39 """Which page is up: the stage's seconds are one page's length, and the pages keep taking 

40 turns for as long as the stage is held.""" 

41 if scene.fb_seconds <= 0: 

42 return 0 

43 return int(max(0.0, scene.fb_elapsed) // scene.fb_seconds) % pages

46def _color(abbrev: str) -> tuple[int, int, int]: 

47 return team_color(abbrev) if is_team(abbrev) else WHITE

50def _logo(r: Renderer, canvas: Any, abbrev: str, cx: int) -> None: 

51 logos = r.football_logos 

52 logo = logos.get(abbrev, 16) if logos is not None and is_team(abbrev) else None 

53 if logo is None: 

54 r.text_center(canvas, r.fonts.small, cx, LOGO_CENTRE_Y + 3, _color(abbrev), abbrev) 

55 return 

56 ox, oy = cx - logo.width // 2, LOGO_CENTRE_Y - logo.height // 2 

57 for dx, dy, (red, green, blue) in logo.pixels: 

58 canvas.SetPixel(ox + dx, oy + dy, red, green, blue)

61def _dots(canvas: Any, cx: int, page: int, pages: int) -> None: 

62 """One 2x2 dot a page under a logo, the current one lit.""" 

63 x0 = cx - (pages * 4 - 2) // 2 

64 for i in range(pages): 

65 shade = LIT_DOT if i == page else UNLIT_DOT 

66 for dx in range(2): 

67 for dy in range(2): 

68 canvas.SetPixel(x0 + i * 4 + dx, 28 + dy, *shade)

71def draw_stats(r: Renderer, canvas: Any, scene: Scene) -> None: 

72 canvas.Clear() 

73 game, stats = scene.fb_game, scene.fb_stats 

74 if game is None or not isinstance(stats, GameStats): 

75 return 

76 away, home = game.away.abbrev, game.home.abbrev 

77 colors = {"away": _color(away), "home": _color(home)} 

78 page = page_of(scene, len(PAGES)) 

79 

80 _logo(r, canvas, away, 8) 

81 _logo(r, canvas, home, r.width - 8) 

82 _dots(canvas, 8, page, len(PAGES)) 

83 _dots(canvas, r.width - 8, page, len(PAGES)) 

84 

85 mid = r.width // 2 

86 for row, baseline in zip(PAGES[page], ROW_BASELINES, strict=False): 

87 ahead = stats.leader(row) 

88 r.text_center(canvas, r.fonts.tiny, mid, baseline, DIM, row.label) 

89 for side, x, draw in ( 

90 ("away", LEFT_X, r.text), 

91 ("home", RIGHT_X, r.text_right), 

92 ): 

93 color = TRAILING if ahead not in (None, side) else colors[side] 

94 draw(canvas, r.fonts.tiny, x, baseline, color, stats.value(side, row.key))
