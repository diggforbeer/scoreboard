1"""The 32 football teams (#347): names, ESPN ids and LED-tuned colours. 

2 

3Kept apart from ``display/teams.py`` on purpose. Football and hockey abbreviations 

4overlap (``BUF``, ``CAR``, ``DAL``, ``DET``, ``MIN`` and others) and mean 

5different teams, so the two tables must never be merged or looked up across. 

6Nothing in here imports the hockey table. 

7 

8Ids, cities and nicknames are ESPN's own (``site.api.espn.com/apis/site/v2/ 

9sports/football/nfl/teams``), pinned by a test. Colours are tuned for an LED 

10matrix the way the hockey ones are: several real primaries are navy, dark green, 

11brown or black and vanish on the panel, so those are substituted with the team's 

12brighter accent or a lifted shade of the same hue. 

13"""

116#: The night games a ranked list can follow as if they were teams: ``TNF``, ``SNF`` and ``MNF`` are 

117#: every game kicking off on a Thursday, Sunday or Monday evening Eastern time. They are not in 

118#: ``TEAMS`` (which is pinned to the 32 clubs); the abbreviations are what ``[[teams]]`` stores. 

119NIGHT_SLOTS: dict[str, str] = { 

120 "TNF": "Thursday Night Football", 

121 "SNF": "Sunday Night Football", 

122 "MNF": "Monday Night Football", 

123} 

124_SLOT_DAY = {"TNF": 3, "SNF": 6, "MNF": 0} # Python weekday: Monday 0 ... Sunday 6 

125#: The league schedules its night games in Eastern time; "evening" is 7pm on. 

126_SLOT_ZONE = "America/New_York" 

127_EVENING_HOUR = 19 

128 

129 

130def is_slot(abbrev: str) -> bool: 

131 return str(abbrev).strip().upper() in NIGHT_SLOTS

134def in_night_slot(slot: str, start_utc: datetime | None) -> bool: 

135 """Whether a game kicking off at ``start_utc`` is in the ``SNF`` / ``MNF`` slot.""" 

136 key = str(slot).strip().upper() 

137 if key not in _SLOT_DAY or start_utc is None: 

138 return False 

139 local = start_utc.astimezone(ZoneInfo(_SLOT_ZONE)) 

140 return local.weekday() == _SLOT_DAY[key] and local.hour >= _EVENING_HOUR

143def team(abbrev: str) -> FootballTeam: 

144 """The team for an abbreviation (case and spaces ignored); raises if unknown.""" 

145 key = str(abbrev).strip().upper() 

146 try: 

147 return TEAMS[key] 

148 except KeyError: 

149 raise UnknownTeamError(f"{abbrev!r} is not a football team abbreviation") from None

152def team_by_id(espn_id: int | str) -> FootballTeam: 

153 """The team for an ESPN team id, given as an int or the string the feed uses.""" 

154 try: 

155 return _BY_ID[int(espn_id)] 

156 except (KeyError, TypeError, ValueError): 

157 raise UnknownTeamError(f"{espn_id!r} is not an ESPN football team id") from None
