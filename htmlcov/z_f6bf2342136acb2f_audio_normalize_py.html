1"""Loudness-normalise an uploaded horn / wah-wah / ding (#322). 

2 

3Uploads used to play at whatever level they were recorded at, and the global 

4volume can't fix a mismatch *between* files. So each upload is decoded to 

516-bit PCM once, scaled to a common RMS level and stored as a mono WAV, which 

6plays with ``aplay`` alone (no ``mpg123`` for uploaded sounds). 

7 

8All the maths is ``array``/``math`` (Python 3.13 dropped ``audioop``). Decoding 

9prefers ``ffmpeg`` when the board has it (any format, one code path); boards 

10flashed before it joined the image fall back to ``wave`` for WAV and 

11``mpg123 -w`` for MP3. Gain stays an RMS match to the default horn rather than 

12ffmpeg's ``loudnorm``: that targets integrated LUFS, a different measure from 

13the RMS the target was taken in, and is unreliable on clips this short. Either 

14way nothing client-supplied reaches a path or a command line. 

15""" 

16

31#: Target loudness, RMS in dBFS. Measured on the shipped default horn 

32#: (assets/horns/_default.wav): RMS 15820 / 32768 = -6.3 dBFS, peak -3.1 dBFS. 

33#: The other shipped sounds sit lower (wah-wah 8415 = -11.8, ding 3874 = 

34#: -18.6) but are short stings; the horn is the loudest thing the board plays 

35#: and the one the owner balances against, so uploads land at its level. 

36TARGET_RMS_DBFS = -6.3 

37#: Gain is capped so the loudest sample stays under this (leaves headroom for 

38#: DAC/resampling overshoot). A peaky recording therefore ends up quieter than 

39#: the target rather than clipping. 

40PEAK_CEILING_DBFS = -1.0 

41#: Below this RMS a file is treated as silent: infinite gain would only 

42#: amplify noise, so it is stored as it is. 

43SILENCE_RMS_DBFS = -60.0 

44#: Longer clips are rejected, not trimmed: cutting a horn mid-sound is worse 

45#: than asking for a shorter file. The default horn is 2.2s. 

46MAX_SECONDS = 10.0 

47#: ``mpg123 -n``: decode at most this many frames (>= 24s of any MP3), so a 

48#: 2 MB upload can't balloon into a huge temp file before the length check. 

49MP3_MAX_FRAMES = 1000 

50MP3_DECODE_TIMEOUT = 20 

51FFMPEG_TIMEOUT = 30 

52 

53_FULL_SCALE = 32768.0 

54_run = subprocess.run # a seam for tests, like GoalHornPlayer._popen 

55_which = shutil.which # likewise: boards flashed before #322 have no ffmpeg 

56_INSTALL = "sudo apt-get install -y ffmpeg" 

57_NEEDS_FFMPEG = f"this format needs ffmpeg, which isn't installed on this board ({_INSTALL})"

73def decode_mp3(raw: bytes) -> bytes: 

74 """MP3 bytes -> WAV bytes via ``mpg123``. Raises ValueError (fit to show).""" 

75 with tempfile.TemporaryDirectory(prefix="horn-") as tmp: 

76 src, dst = Path(tmp) / "in.mp3", Path(tmp) / "out.wav" 

77 src.write_bytes(raw) 

78 cmd = ["mpg123", "-q", "-n", str(MP3_MAX_FRAMES), "-w", str(dst), str(src)] 

79 try: 

80 proc = _run(cmd, capture_output=True, timeout=MP3_DECODE_TIMEOUT, check=False) 

81 except FileNotFoundError: 

82 raise ValueError( 

83 "MP3 uploads need mpg123, which isn't installed on this board " 

84 "(sudo apt install mpg123), or upload a WAV instead" 

85 ) from None 

86 except subprocess.TimeoutExpired: 

87 raise ValueError("the MP3 took too long to decode") from None 

88 if proc.returncode != 0 or not dst.is_file() or dst.stat().st_size == 0: 

89 raise ValueError("could not decode the MP3 file") 

90 return dst.read_bytes()

93def decode_ffmpeg(raw: bytes) -> bytes: 

94 """Any audio format -> mono 16-bit WAV bytes via ``ffmpeg`` (#322). 

95 

96 Output is cut at ``MAX_SECONDS`` + 1 so an over-long clip is still seen as 

97 too long without decoding a whole 2 MB upload. Fixed argv, fixed file 

98 names in a temp directory: nothing client-supplied reaches the command. 

99 """ 

100 with tempfile.TemporaryDirectory(prefix="horn-") as tmp: 

101 src, dst = Path(tmp) / "in.bin", Path(tmp) / "out.wav" 

102 src.write_bytes(raw) 

103 cmd = [ 

104 "ffmpeg", "-nostdin", "-v", "error", "-y", "-i", str(src), 

105 "-vn", "-t", str(MAX_SECONDS + 1), "-ac", "1", "-c:a", "pcm_s16le", 

106 "-f", "wav", str(dst), 

107 ] # fmt: skip 

108 try: 

109 proc = _run(cmd, capture_output=True, timeout=FFMPEG_TIMEOUT, check=False) 

110 except FileNotFoundError: 

111 raise ValueError(_NEEDS_FFMPEG) from None 

112 except subprocess.TimeoutExpired: 

113 raise ValueError("the audio file took too long to decode") from None 

114 if proc.returncode != 0 or not dst.is_file() or dst.stat().st_size == 0: 

115 raise ValueError("could not decode the audio file") 

116 return dst.read_bytes()

146def _read_wav(data: bytes) -> tuple[array.array, int]: 

147 """Mono 16-bit samples and the sample rate. ValueError if unreadable.""" 

148 try: 

149 with wave.open(io.BytesIO(data)) as wav: 

150 channels, width, rate = wav.getnchannels(), wav.getsampwidth(), wav.getframerate() 

151 if wav.getnframes() <= 0: 

152 raise ValueError("WAV file has no audio")

156 frames = wav.readframes(min(wav.getnframes(), int(MAX_SECONDS * rate) + rate))

182def normalize(raw: bytes, is_mp3: bool) -> Normalized: 

183 """Decode, measure and rescale an upload. Raises ValueError (fit to show).""" 

184 if _which("ffmpeg"): 

185 wav_in = decode_ffmpeg(raw) 

186 elif is_mp3: 

187 wav_in = decode_mp3(raw) 

188 elif _is_wav(raw): 

189 wav_in = raw 

190 else: 

191 raise ValueError(f"not a valid WAV or MP3 file (other formats need ffmpeg: {_INSTALL})") 

192 samples, rate = _read_wav(wav_in) 

193 seconds = len(samples) / rate 

194 if seconds > MAX_SECONDS: 

195 raise ValueError(f"the clip is too long ({seconds:.0f}s, max {MAX_SECONDS:.0f}s)") 

196 

197 rms = math.sqrt(sum(s * s for s in samples) / len(samples)) 

198 peak = max(max(samples), -min(samples)) 

199 if rms / _FULL_SCALE < 10 ** (SILENCE_RMS_DBFS / 20): 

200 # Keep a WAV exactly as sent; an MP3 still has to be stored as WAV. 

201 wav = raw if wav_in is raw else _wav_bytes(samples, rate) 

202 return Normalized(wav, None, "stored unchanged: the file is silent or nearly silent") 

203 

204 want = 10 ** (TARGET_RMS_DBFS / 20) * _FULL_SCALE / rms 

205 ceiling = 10 ** (PEAK_CEILING_DBFS / 20) * _FULL_SCALE / peak 

206 gain = min(want, ceiling) 

207 scaled = array.array("h", [max(-32768, min(32767, round(s * gain))) for s in samples]) 

208 gain_db = round(_db(gain), 1) 

209 note = f"normalized, {gain_db:+.1f} dB" 

210 if ceiling < want: 

211 note += " (limited by peak level, so it stays a little quieter than the target)" 

212 return Normalized(_wav_bytes(scaled, rate), gain_db, note)
