38def is_thanksgiving(day: date) -> bool: 

39 """The fourth Thursday of November.""" 

40 return day.month == 11 and day.weekday() == 3 and 22 <= day.day <= 28

43def game_holiday(start_utc: datetime | None, tz: tzinfo) -> str | None: 

44 """``thanksgiving`` or ``christmas`` for a game on that day in the board's own zone, else 

45 ``None``: the only two days a game screen gets a holiday decoration.""" 

46 if start_utc is None: 

47 return None 

48 day = start_utc.astimezone(tz).date() 

49 if is_thanksgiving(day): 

50 return "thanksgiving" 

51 return "christmas" if day.month == 12 and day.day == 25 else None

54def game_tag(start_utc: datetime | None, season_type: int | None = None) -> str: 

55 """``THANKSGIVING``, ``CHRISTMAS``, ``THU NIGHT``, ``MON NIGHT`` or ``''``, by the game's 

56 Eastern date and hour. Preseason games carry none.""" 

57 if start_utc is None or season_type == PRESEASON: 

58 return "" 

59 local = start_utc.astimezone(EASTERN) 

60 day = local.date() 

61 if is_thanksgiving(day): 

62 return "THANKSGIVING" 

63 if day.month == 12 and day.day == 25: 

64 return "CHRISTMAS" 

65 if local.hour >= NIGHT_HOUR: 

66 return {3: "THU NIGHT", 0: "MON NIGHT"}.get(day.weekday(), "") 

67 return ""

70def week_label( 

71 season_type: int | None, 

72 week: int | None, 

73 start_utc: datetime | None, 

74 calendar: tuple[CalendarEntry, ...] = (), 

75) -> str: 

76 """``WEEK 11``, ``PRESEASON WK 2``, or the playoff round (``WILD CARD``, ``DIVISIONAL``, 

77 ``CONF CHAMP``, ``SUPER BOWL``) read from the calendar entry the game's date falls in.""" 

78 if season_type == POSTSEASON: 

79 for entry in calendar: 

80 if ( 

81 entry.season_type == POSTSEASON 

82 and start_utc is not None 

83 and entry.start is not None 

84 and entry.end is not None 

85 and entry.start <= start_utc < entry.end 

86 ): 

87 return ROUND_LABELS.get(entry.short_label, entry.short_label.upper()) 

88 return "PLAYOFFS" 

89 if week is None: 

90 return "" 

91 return f"PRESEASON WK {week}" if season_type == PRESEASON else f"WEEK {week}"

114def _find_event(ctx: IdleContext, away: str, home: str, start: datetime) -> FbGame | None: 

115 """The scoreboard's own event for this game (it has the records), if the board has one.""" 

116 pools = [ctx.games] 

117 week = ctx.idle.week(ctx.now) 

118 if week is not None: 

119 pools.append(week.games) 

120 for pool in pools: 

121 for game in pool: 

122 if ( 

123 game.away.abbrev == away 

124 and game.home.abbrev == home 

125 and game.start_utc is not None 

126 and abs(game.start_utc - start) < timedelta(days=1) 

127 ): 

128 return game 

129 return None

132def build_preview(ctx: IdleContext) -> PreviewView | None: 

133 """The favourite's next game, or None (no favourite, nothing scheduled, a game not yet set). 

134 

135 Candidates are the favourite's not-yet-started game on the scoreboard (the only place a 

136 preseason or playoff game appears) and the next unplayed game of its season schedule; the 

137 earlier wins. A ``TBD @ TBD`` playoff game is skipped until its teams are known. Records come 

138 from the scoreboard event when the board has one, and are left out otherwise. 

139 """ 

140 favourite = ctx.config.favourite_team 

141 if not favourite or not _known(favourite): 

142 return None 

143 now = ctx.utc 

144 found: list[tuple[datetime, str, str, int | None, int | None]] = [] 

145 for g in ctx.games: 

146 playable = ( 

147 g.involves(favourite) 

148 and g.is_pregame 

149 and g.start_utc is not None 

150 and g.start_utc >= now 

151 ) 

152 if playable and _known(g.away.abbrev) and _known(g.home.abbrev): 

153 found.append((g.start_utc, g.away.abbrev, g.home.abbrev, g.season_type, g.week)) 

154 schedule = ctx.idle.schedule(team(favourite).espn_id, ctx.now) 

155 upcoming = schedule.next_game(now) if schedule is not None else None 

156 if upcoming is not None and upcoming.start_utc is not None: 

157 found.append( 

158 (upcoming.start_utc, upcoming.away, upcoming.home, upcoming.season_type, upcoming.week) 

159 ) 

160 if not found: 

161 return None 

162 # Earliest first; on a tie the scoreboard's entry (listed first) wins. 

163 start, away, home, season_type, week = min(found, key=lambda f: f[0]) 

164 event = _find_event(ctx, away, home, start) 

165 calendar = ctx.idle.scoreboard.calendar if ctx.idle.scoreboard is not None else () 

166 if event is not None: 

167 season_type = event.season_type if event.season_type is not None else season_type 

168 week = event.week if event.week is not None else week 

169 return PreviewView( 

170 away=away, 

171 home=home, 

172 start_utc=start, 

173 tag=game_tag(start, season_type), 

174 week=week_label(season_type, week, start, calendar), 

175 away_record=event.away.record if event is not None else "", 

176 home_record=event.home.record if event is not None else "", 

177 )

199#: Suffixes that are not a surname: ``Odell Beckham Jr.`` is BECKHAM.

205def surname(name: str) -> str: 

206 """``J.J. McCarthy`` -> ``MCCARTHY``; a trailing ``Jr.`` or ``III`` is skipped.""" 

207 parts = [p for p in name.split() if p.upper() not in _SUFFIXES] 

208 return (parts[-1] if parts else name).upper()

234def margin_line(game: FbGame) -> str: 

235 gap = game.home.score - game.away.score 

236 if gap == 0: 

237 return "TIE GAME" 

238 winner = game.home if gap > 0 else game.away 

239 return f"{winner.abbrev} WIN BY {abs(gap)}"

242def recap_lines(game: FbGame) -> tuple[RecapLine, ...]: 

243 """The game's leaders with a yard count, in the recap's order, one per category. A leader 

244 without a readable yard count, or with a team that is not on this game, is dropped.""" 

245 out: list[RecapLine] = [] 

246 for category in RECAP_CATEGORIES: 

247 leader = next((lead for lead in game.leaders if lead.category == category), None) 

248 if leader is None or leader.team_id is None: 

249 continue 

250 match = re.search(r"(\d+)\s*YDS?", leader.display.upper()) 

251 try: 

252 abbrev = team_by_id(leader.team_id).abbrev 

253 except KeyError: 

254 continue 

255 if match is None or abbrev not in (game.home.abbrev, game.away.abbrev): 

256 continue 

257 out.append(RecapLine(abbrev, surname(leader.name), int(match.group(1)), category)) 

258 return tuple(out)

261def build_final(ctx: IdleContext) -> FinalView | None: 

262 """The favourite's game if it is over; ``None`` otherwise. (How long it stays up is 

263 ``FootballMode``'s hold, not this builder's.)""" 

264 favourite = ctx.config.favourite_team 

265 game = next((g for g in ctx.games if favourite and g.involves(favourite) and g.is_final), None)

305def build_standings(ctx: IdleContext) -> StandingsView | None: 

306 """The standings pages for the favourite, or None (no favourite, no table yet, or a favourite 

307 the table does not list).""" 

308 favourite = ctx.config.favourite_team.strip().upper() 

309 table = ctx.idle.standings(ctx.now) 

310 if not favourite or table is None: 

311 return None 

312 division = table.division_of(favourite) 

313 conference = table.conference_of(favourite) 

314 if division is None or conference is None: 

315 return None 

316 seeded = conference.seeds() 

317 out = seeded[PLAYOFF_SEEDS] if len(seeded) > PLAYOFF_SEEDS else None 

318 if out is None: 

319 rest = conference.unseeded() 

320 out = rest[0] if rest else None 

321 board = ctx.idle.scoreboard 

322 week = f"WK {board.week}" if board is not None and board.week is not None else "" 

323 return StandingsView( 

324 team=favourite, 

325 division_name=division.name.upper(), 

326 division=division.rows, 

327 conference=conference.abbreviation.upper(), 

328 seeds=seeded[:PLAYOFF_SEEDS], 

329 first_out=out, 

330 week=week, 

331 )

354def week_order(games: Sequence[FbGame]) -> tuple[FbGame, ...]: 

355 """Live games first (the ones to watch), then those still to come by kickoff, then the finals 

356 by kickoff. A game with no kickoff time goes last of its group.""" 

357 far = datetime.max.replace(tzinfo=UTC) 

358 

359 def key(game: FbGame) -> tuple[int, datetime]: 

360 group = 0 if game.is_live else 1 if game.is_pregame else 2 

361 return group, game.start_utc or far 

362 

363 return tuple(sorted(games, key=key)[:WEEK_MAX_GAMES])

366def build_week(ctx: IdleContext) -> WeekView | None: 

367 """The week's slate: the whole-week fetch when the board has one, else today's scoreboard.

394def build_schedule(ctx: IdleContext) -> ScheduleView | None: 

395 """The next three games from the favourite's schedule. A game with no week number (the feed's 

396 bye) or no kickoff time, or against a team not yet known, is passed over; fewer than three left 

397 shows what is left; nothing left (or no schedule yet) is ``None`` so the screen is skipped.""" 

398 favourite = ctx.config.favourite_team.strip().upper() 

399 if not favourite or not _known(favourite): 

400 return None 

401 schedule = ctx.idle.schedule(team(favourite).espn_id, ctx.now) 

402 if schedule is None: 

403 return None 

404 rows = [] 

405 for g in schedule.upcoming(ctx.utc): 

406 opponent = g.opponent(favourite) 

407 if g.week is None or g.start_utc is None or not _known(opponent): 

408 continue 

409 rows.append(ScheduleRow(g.week, g.start_utc, opponent, g.is_home(favourite))) 

410 if len(rows) == SCHEDULE_ROWS: 

411 break 

412 return ScheduleView(favourite, tuple(rows)) if rows else None
